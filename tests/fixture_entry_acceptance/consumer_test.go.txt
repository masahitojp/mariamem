var fixture *mariamemtest.Fixture
var prepares int

func migrate(ctx context.Context, dsn, mode string) error {
	binary, err := os.Executable()
	if err != nil {
		return err
	}
	cmd := exec.CommandContext(ctx, binary, "-test.run=^$")
	cmd.Env = append(os.Environ(), "MARIAMEM_G1_MIGRATE="+mode, "MARIAMEM_G1_DSN="+dsn)
	output, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("migration command: %w; stderr: %s", err, output)
	}
	return nil
}

func migrationCommand() int {
	if os.Getenv("MARIAMEM_G1_MIGRATE") == "fail" {
		fmt.Fprintln(os.Stderr, "CONTROLLED_MIGRATION_FAILURE")
		return 23
	}
	p, err := sql.Open("mysql", os.Getenv("MARIAMEM_G1_DSN"))
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		return 2
	}
	for _, q := range []string{"CREATE TABLE users (id INT PRIMARY KEY, label VARCHAR(32))", "INSERT INTO users VALUES (1,'seed')"} {
		if _, err = p.Exec(q); err != nil {
			p.Close()
			fmt.Fprintln(os.Stderr, err)
			return 2
		}
	}
	if err = p.Close(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		return 2
	}
	return 0
}

func TestMain(m *testing.M) {
	if os.Getenv("MARIAMEM_G1_MIGRATE") != "" {
		os.Exit(migrationCommand())
	}
	code := mariamemtest.Run(m, mariamemtest.Options{MaxForks: 2,
		Prepare: func(ctx context.Context, _ *sql.DB, dsn string) error { prepares++; return migrate(ctx, dsn, "ok") },
	}, func(f *mariamemtest.Fixture) { fixture = f })
	if prepares != 1 {
		fmt.Fprintln(os.Stderr, "preparation count", prepares)
		code = 1
	}
	result := "PASS_EXTERNAL_FIXTURE"
	if code != 0 {
		result = "FAIL_EXTERNAL_FIXTURE"
	}
	json.NewEncoder(os.Stdout).Encode(map[string]any{"result": result, "prepares": prepares})
	os.Exit(code)
}

func seed(t *testing.T, p *sql.DB) {
	t.Helper()
	var label string
	if err := p.QueryRowContext(t.Context(), "SELECT label FROM users WHERE id=?", 1).Scan(&label); err != nil || label != "seed" {
		t.Fatalf("seed %q: %v", label, err)
	}
	var n int
	if err := p.QueryRowContext(t.Context(), "SELECT COUNT(*) FROM information_schema.COLUMNS WHERE TABLE_SCHEMA=DATABASE() AND TABLE_NAME='users'").Scan(&n); err != nil || n != 2 {
		t.Fatalf("schema %d: %v", n, err)
	}
}
func TestFirstCommit(t *testing.T) {
	p := fixture.DB(t)
	seed(t, p)
	if p != fixture.DB(t) || p.Stats().MaxOpenConnections != 2 {
		t.Fatal("pool identity/budget")
	}
	for _, q := range []string{"DELETE FROM users", "ALTER TABLE users ADD changed INT"} {
		if _, err := p.Exec(q); err != nil {
			t.Fatal(err)
		}
	}
}
func TestSecondSeed(t *testing.T) { seed(t, fixture.DB(t)) }
func TestParallel(t *testing.T) {
	for i := 0; i < 4; i++ {
		t.Run(fmt.Sprint(i), func(t *testing.T) {
			t.Parallel()
			p := fixture.DB(t)
			seed(t, p)
			if _, err := p.Exec("DELETE FROM users"); err != nil {
				t.Fatal(err)
			}
		})
	}
}
func TestOrdinaryFailureDoesNotContaminate(t *testing.T) {
	if os.Getenv("MARIAMEM_G1_FAIL_BODY") == "1" {
		t.Run("intentional_failure", func(t *testing.T) {
			p := fixture.DB(t)
			seed(t, p)
			if _, err := p.Exec("DELETE FROM users"); err != nil {
				t.Fatal(err)
			}
			t.Fatal("CONTROLLED_BODY_FAILURE")
		})
		t.Run("healthy", func(t *testing.T) { seed(t, fixture.DB(t)); t.Log("HEALTHY_SEED_AFTER_FAILURE") })
		return
	}
	binary, err := os.Executable()
	if err != nil {
		t.Fatal(err)
	}
	cmd := exec.CommandContext(t.Context(), binary, "-test.run=^TestOrdinaryFailureDoesNotContaminate$", "-test.v")
	cmd.Env = append(os.Environ(), "MARIAMEM_G1_FAIL_BODY=1")
	out, err := cmd.CombinedOutput()
	var exit *exec.ExitError
	if !errors.As(err, &exit) || exit.ExitCode() != 1 || !strings.Contains(string(out), "CONTROLLED_BODY_FAILURE") || !strings.Contains(string(out), "--- PASS: TestOrdinaryFailureDoesNotContaminate/healthy") || !strings.Contains(string(out), "HEALTHY_SEED_AFTER_FAILURE") {
		t.Fatalf("failed-body oracle: %v\n%s", err, out)
	}
}
func noTemps(t *testing.T, path string) {
	t.Helper()
	entries, err := os.ReadDir(path)
	if err != nil || len(entries) != 0 {
		t.Fatalf("temporary owners retained: %v %v", entries, err)
	}
}
func TestMigrationFailure(t *testing.T) {
	temp := t.TempDir()
	t.Setenv("TMPDIR", temp)
	f, err := mariamemtest.New(t.Context(), mariamemtest.Options{Prepare: func(ctx context.Context, _ *sql.DB, dsn string) error { return migrate(ctx, dsn, "fail") }})
	var command *exec.ExitError
	var detail *mariamemtest.Error
	if f != nil || !errors.As(err, &command) || command.ExitCode() != 23 || !errors.As(err, &detail) || detail.Stage != "prepare" || !strings.Contains(err.Error(), "CONTROLLED_MIGRATION_FAILURE") {
		t.Fatal("command identity/stderr/admission", err)
	}
	noTemps(t, temp)
}
func TestCancelledPrepare(t *testing.T) {
	temp := t.TempDir()
	t.Setenv("TMPDIR", temp)
	ctx, cancel := context.WithCancel(t.Context())
	defer cancel()
	f, err := mariamemtest.New(ctx, mariamemtest.Options{Prepare: func(ctx context.Context, _ *sql.DB, _ string) error { cancel(); return ctx.Err() }})
	if f != nil || !errors.Is(err, context.Canceled) {
		t.Fatal("cancellation/admission", err)
	}
	noTemps(t, temp)
}
