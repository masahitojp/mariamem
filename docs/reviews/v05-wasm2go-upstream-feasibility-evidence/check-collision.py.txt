import argparse, hashlib, os, sys
from pathlib import Path
root=Path.cwd()
sys.path.insert(0,str(root/'scripts'))
from build_feedback import go_environment,run
p=argparse.ArgumentParser();p.add_argument('--converter',type=Path,required=True);p.add_argument('--output',type=Path,required=True);a=p.parse_args()
out=a.output.resolve();out.mkdir(parents=True,exist_ok=False)
(out/'go.mod').write_text('module example.com/collision\n\ngo 1.26.0\n')
(out/'collision.wasm').write_bytes(bytes([0,97,115,109,1,0,0,0,1,5,1,96,0,1,127,3,3,2,0,0,7,20,2,6,88,65,108,108,111,99,0,0,7,120,95,97,108,108,111,99,0,1,10,11,2,4,0,65,1,11,4,0,65,2,11]))
(out/'fixture').mkdir()
env=go_environment(dict(os.environ,GOTOOLCHAIN='go1.26.8',GOWORK='off',GOENV='off',GOFLAGS='',GOEXPERIMENT=''))
args=[str(a.converter.resolve()),'-pure','-i',str(out/'collision.wasm'),'-o',str(out/'fixture/fixture.go'),'-pkg','fixture','-import','example.com/collision/fixture']
run(args,env=env,check=True,phase='upstream:collision-fixture-translation')
first=(out/'fixture/fixture.go').read_bytes()
run(args,env=env,check=True,phase='upstream:collision-fixture-retranslation')
assert first==(out/'fixture/fixture.go').read_bytes(),'non-deterministic wrapper output'
(out/'fixture/collision_test.go').write_text('package fixture\nimport "testing"\nfunc TestDistinctCollidingExports(t *testing.T) {m:=New();if m.XAlloc()!=1 || m.XAlloc_x_alloc()!=2 {t.Fatal("lost/aliased export")}}\n')
run(['go','test','-p','1','./fixture','-run','TestDistinctCollidingExports$','-count=1','-v'],cwd=out,env=env,check=True,phase='upstream:collision-distinct-exports',inputs=[a.converter,out/'collision.wasm',out/'fixture/fixture.go',out/'fixture/collision_test.go'])
print('PASS: deterministic wrapper source; distinct export return values 1 and 2; generated SHA256',hashlib.sha256(first).hexdigest())
