// Align both transports at the same partial request (16 KiB sent) before
// requesting low-level Shutdown. This barrier is independent of OS pipe size.
func TestQualificationPartialControl(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
	defer cancel()
	s, err := StartGenerated(ctx, nil, 6*time.Second, io.Discard)
	if err != nil {
		t.Fatal(err)
	}
	defer s.Close(ctx)
	p := s.Guest
	r, err := p.Call(ctx, 1, 0, nil)
	if err != nil || !r.OK {
		t.Fatal(err, r.Error)
	}
	hit, release := p.QualPartial()
	defer release()
	result := make(chan error, 1)
	go func() {
		r, e := p.Call(ctx, 2, 0, []byte("SELECT 'q' /*"+strings.Repeat("a", 131072)+"*/"))
		if e == nil {
			if !r.OK || len(r.Rows) != 1 || len(r.Rows[0]) != 1 || r.Rows[0][0] == nil {
				e = fmt.Errorf("bad SQL response")
			} else {
				v, decodeErr := hex.DecodeString(r.Rows[0][0].Hex)
				if decodeErr != nil || string(v) != "q" {
					e = fmt.Errorf("bad bytes")
				}
			}
		}
		result <- e
	}()
	waitFull(t, hit)
	shut := make(chan error, 1)
	go func() { shut <- p.Shutdown(ctx) }()
	until := time.Now().Add(2 * time.Second)
	for !p.QualStopping() {
		if time.Now().After(until) {
			t.Fatal("shutdown not started")
		}
		runtime.Gosched()
	}
	release()
	queryErr := waitFull(t, result)
	if queryErr != nil && !errors.Is(queryErr, io.EOF) && !strings.Contains(queryErr.Error(), "guest exited") {
		t.Fatal("unexpected error", queryErr)
	}
	shutErr := waitFull(t, shut)
	waitFull(t, p.Done())
	e := s.Close(ctx)
	if e != nil || !s.DisposalComplete() {
		t.Fatal("disposal", e)
	}
	t.Logf("partial_sent=16384 transport=%s query_err=%v shutdown_err=%v disposed=true", os.Getenv("MARIAMEM_BENCH_STREAM"), queryErr, shutErr)
}

func TestQualificationClientDrain(t *testing.T) {
	for _, direction := range []string{"input", "output"} {
		t.Run(direction, func(t *testing.T) {
			ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
			defer cancel()
			s, err := StartGenerated(ctx, nil, 6*time.Second, io.Discard)
			if err != nil {
				t.Fatal(err)
			}
			defer s.Close(ctx)
			cfg := mysql.NewConfig()
			cfg.User = "root"
			cfg.DBName = "test"
			cfg.Net = "tcp"
			cfg.Addr = "127.0.0.1:1"
			cfg.DialFunc = func(ctx context.Context, _, _ string) (net.Conn, error) { return s.DialContext(ctx) }
			connector, err := mysql.NewConnector(cfg)
			if err != nil {
				t.Fatal(err)
			}
			db := sql.OpenDB(connector)
			defer db.Close()
			conn, err := db.Conn(ctx)
			if err != nil {
				t.Fatal(err)
			}
			defer conn.Close()
			var one int
			if e := conn.QueryRowContext(ctx, "SELECT 1").Scan(&one); e != nil || one != 1 {
				t.Fatal(e)
			}
			stmt := "SELECT REPEAT('q',131072)"
			want := strings.Repeat("q", 131072)
			if direction == "input" {
				stmt = "SELECT 'q' /*" + strings.Repeat("a", 131072) + "*/"
				want = "q"
			}
			hit, release := s.Guest.QualPause(direction)
			defer release()
			result := make(chan error, 1)
			go func() {
				var v string
				e := conn.QueryRowContext(ctx, stmt).Scan(&v)
				if e == nil && v != want {
					e = fmt.Errorf("content mismatch")
				}
				result <- e
			}()
			waitFull(t, hit)
			release()
			if e := waitFull(t, result); e != nil {
				t.Fatal(e)
			}
			if e := conn.Close(); e != nil {
				t.Fatal(e)
			}
			if e := db.Close(); e != nil {
				t.Fatal(e)
			}
			for {
				active, changed, failed, done := s.DisconnectState()
				if active == 0 {
					break
				}
				select {
				case <-changed:
				case <-failed:
					t.Fatal("unexpected failure")
				case <-done:
					t.Fatal("early exit")
				case <-ctx.Done():
					t.Fatal(ctx.Err())
				}
			}
			if e := s.Guest.Err(); e != nil {
				t.Fatal("guest unhealthy before close", e)
			}
			if e := s.Close(ctx); e != nil {
				t.Fatal(e)
			}
			waitFull(t, s.Guest.Done())
			if !s.DisposalComplete() {
				t.Fatal("disposal")
			}
			if _, e := os.Stat(s.transfer); !os.IsNotExist(e) {
				t.Fatal("transfer retained", e)
			}
			t.Logf("observed_full_bytes=16384 client_payload_verified=%d graceful_close=true", len(want))
		})
	}
}
