func waitFull[T any](t *testing.T, ch <-chan T) T {
	t.Helper()
	select {
	case v := <-ch:
		return v
	case <-time.After(4 * time.Second):
		t.Fatal("operation did not complete within 4s")
	}
	var zero T
	return zero
}

func TestQualificationFullStream(t *testing.T) {
	for _, direction := range []string{"input", "output"} {
		for _, event := range []string{"cancel", "disconnect", "drain", "serverClose"} {
			t.Run(direction+"/"+event, func(t *testing.T) { qualFullCase(t, direction, event) })
		}
	}
}

func qualFullCase(t *testing.T, direction, event string) {
	ctx, cancel := context.WithTimeout(context.Background(), 12*time.Second)
	defer cancel()
	s, err := StartGenerated(ctx, nil, 8*time.Second, io.Discard)
	if err != nil {
		t.Fatal(err)
	}
	p := s.Guest
	closed := false
	defer func() {
		if !closed {
			c, done := context.WithTimeout(context.Background(), 4*time.Second)
			defer done()
			_ = s.Close(c)
		}
	}()
	query := "SELECT REPEAT('q',131072)"
	want := strings.Repeat("q", 131072)
	if direction == "input" {
		query = "SELECT 'q' /*" + strings.Repeat("a", 131072) + "*/"
		want = "q"
	}
	var wire net.Conn
	var conn *sql.Conn
	var db *sql.DB
	network := event == "disconnect" || event == "serverClose"
	if network {
		cfg := mysql.NewConfig()
		cfg.User = "root"
		cfg.DBName = "test"
		cfg.Net = "tcp"
		cfg.Addr = "127.0.0.1:1"
		cfg.DialFunc = func(ctx context.Context, _, _ string) (net.Conn, error) {
			c, e := s.DialContext(ctx)
			if e == nil {
				wire = c
			}
			return c, e
		}
		connector, e := mysql.NewConnector(cfg)
		if e != nil {
			t.Fatal(e)
		}
		db = sql.OpenDB(connector)
		defer db.Close()
		conn, e = db.Conn(ctx)
		if e != nil {
			t.Fatal(e)
		}
		defer conn.Close()
		var one int
		if e = conn.QueryRowContext(ctx, "SELECT 1").Scan(&one); e != nil || one != 1 {
			t.Fatal("handshake/query", e, one)
		}
	} else {
		opened, e := p.Call(ctx, 1, 0, nil)
		if e != nil || !opened.OK {
			t.Fatal("session open", e, opened.Error)
		}
	}
	full, release := p.QualPause(direction)
	defer release()
	callCtx, stopCall := context.WithCancel(ctx)
	defer stopCall()
	result := make(chan error, 1)
	go func() {
		if network {
			var got string
			e := conn.QueryRowContext(callCtx, query).Scan(&got)
			if e == nil && got != want {
				e = fmt.Errorf("response mismatch")
			}
			result <- e
			return
		}
		r, e := p.Call(callCtx, 2, 0, []byte(query))
		if e == nil {
			if !r.OK || len(r.Rows) != 1 || len(r.Rows[0]) != 1 || r.Rows[0][0] == nil {
				e = fmt.Errorf("unexpected SQL result: %s", r.Error)
			} else {
				b, decodeErr := hex.DecodeString(r.Rows[0][0].Hex)
				if decodeErr != nil || string(b) != want {
					e = fmt.Errorf("returned bytes differ")
				}
			}
		}
		result <- e
	}()
	waitFull(t, full)
	start := time.Now()
	switch event {
	case "cancel":
		stopCall()
		if e := waitFull(t, result); !errors.Is(e, context.Canceled) {
			t.Fatal("cancelled call", e)
		}
		if !errors.Is(p.Err(), context.Canceled) {
			t.Fatal("guest cause", p.Err())
		}
	case "disconnect", "serverClose":
		closeResult := make(chan error, 1)
		if event == "disconnect" {
			if e := wire.Close(); e != nil {
				t.Fatal(e)
			}
		} else {
			go func() { closeResult <- s.Close(ctx) }()
		}
		if e := waitFull(t, result); e == nil {
			t.Fatal("disconnected client succeeded")
		}
		waitFull(t, p.FailureDone())
		if e := p.Err(); e == nil || !strings.Contains(e.Error(), "disconnected during query") {
			t.Fatal("monitor cause", e)
		}
		if event == "serverClose" {
			if e := waitFull(t, closeResult); e != nil {
				t.Fatal("busy close", e)
			}
			closed = true
		}
	case "drain":
		ended := make(chan error, 1)
		go func() { ended <- p.Shutdown(ctx) }()
		until := time.Now().Add(2 * time.Second)
		for !p.QualStopping() {
			if time.Now().After(until) {
				t.Fatal("shutdown did not start")
			}
			runtime.Gosched()
		}
		release()
		if e := waitFull(t, result); e != nil {
			t.Fatal("drain pending response", e)
		}
		if e := waitFull(t, ended); e != nil {
			t.Fatal("shutdown", e)
		}
		if e := p.WaitStopped(ctx); e != nil {
			t.Fatal("stopped", e)
		}
	}
	waitFull(t, p.Done())
	if !closed {
		if e := s.Close(ctx); e != nil {
			t.Fatal("server close", e)
		}
		closed = true
	}
	if e := p.RetryDisposal(ctx); e != nil || !s.DisposalComplete() {
		t.Fatal("disposal", e, s.DisposalComplete())
	}
	if _, e := os.Stat(s.transfer); !os.IsNotExist(e) {
		t.Fatal("retained transfer", e)
	}
	s.mu.Lock()
	clients := len(s.clients)
	s.mu.Unlock()
	if clients != 0 {
		t.Fatal("retained clients", clients)
	}
	t.Logf("observed_full_bytes=16384 event=%s done_and_disposed_us=%d", event, time.Since(start).Microseconds())
}
