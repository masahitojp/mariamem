func TestRsegCostNestedTimingAndCounts(t *testing.T){
 now:=time.Unix(1,0);r:=&RsegCost{now:func()time.Time{return now}}
 r.Register();r.ScanBegin(16384);now=now.Add(10*time.Nanosecond)

}
