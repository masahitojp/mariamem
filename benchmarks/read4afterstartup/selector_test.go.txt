package base
import("bytes";"encoding/binary";"sync/atomic";"testing")
func TestReadSetting(t *testing.T) {
 oldR,oldP:=readThreads,poolMiB;defer func(){readThreads,poolMiB=oldR,oldP}();poolMiB="16"
 fixture:=func()*Module {b:=bytes.Repeat([]byte{0xa5},16384);for i,s:=range expectedSettingsArgs {p:=4096+i*256;copy(b[p:],append([]byte(s),0));binary.LittleEndian.PutUint32(b[128+i*4:],uint32(p))};m:=&Module{Memory:b,MemSize:&atomic.Uint64{}};m.MemSize.Store(uint64(len(b)));return m}
 for _,v:=range []string{"1","4","0","8","bad",""} {t.Run(v,func(t *testing.T){readThreads=v;m:=fixture();before:=bytes.Clone(m.Memory);ok:=v=="1"||v=="4";panicked:=false;func(){defer func(){if recover()!=nil{panicked=true}}();PoolArgv(m,25,128)}();if panicked==ok{t.Fatal("validation")};want:=bytes.Clone(before);if v=="4"{want[4096+16*256+len(expectedSettingsArgs[16])-1]='4'};if !bytes.Equal(m.Memory,want){t.Fatal("unexpected mutation")}})}
}
func TestReadSettingInvalidPreimage(t *testing.T) {
 oldR,oldP:=readThreads,poolMiB;defer func(){readThreads,poolMiB=oldR,oldP}()
 for _,bad:=range []string{"pool","argc","first","last"}{t.Run(bad,func(t *testing.T){
  readThreads,poolMiB="4","16"
  b:=bytes.Repeat([]byte{0xa5},16384);for i,s:=range expectedSettingsArgs {p:=4096+i*256;copy(b[p:],append([]byte(s),0));binary.LittleEndian.PutUint32(b[128+i*4:],uint32(p))}
  m:=&Module{Memory:b,MemSize:&atomic.Uint64{}};m.MemSize.Store(uint64(len(b)));argc:=int32(25)
  switch bad {case "pool":poolMiB="8";case "argc":argc=24;case "first":m.Memory[4096]='X';case "last":m.Memory[4096+24*256]='X'}
  before:=bytes.Clone(m.Memory);func(){defer func(){if recover()==nil{t.Fatal("invalid accepted")}}();PoolArgv(m,argc,128)}()
  if !bytes.Equal(before,m.Memory){t.Fatal("partial mutation")}
 })}
}
