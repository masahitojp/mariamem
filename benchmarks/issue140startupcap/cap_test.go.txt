package base
import "testing"
func TestStartupCap(t *testing.T){
 startupMu.Lock();oldDB,oldPending,oldEvents:=startupDB,startupPending,startupEvents;startupDB=map[*ThreadPool]string{};startupPending=map[*Module]*StartupCapEvent{};startupEvents=nil;startupMu.Unlock();defer func(){startupDB,startupPending,startupEvents=oldDB,oldPending,oldEvents}()
 for _,mode:=range []string{"restricted","normal"}{for _,phase:=range []string{"load","recovery"}{t.Run(mode+phase,func(t *testing.T){t.Setenv("ISSUE140_STARTUP_CAP",mode);m:=&Module{Threads:&ThreadPool{}};RegisterStartupCap(m,"control");got:=StartupCapArg(m,128,1,phase);want:=int32(1);if mode=="normal"{want=0};if got!=want{t.Fatal("argument")};ResolveStartupCap(m,128,16,1,500);v:=startupEvents[len(startupEvents)-1];if v.Selected!=want||v.Resolved!=16||v.Min!=1||v.Max!=500||len(startupPending)!=0{t.Fatal("resolution")}})}}
 for _,bad:=range []string{"mode","request","phase"}{t.Run(bad,func(t *testing.T){t.Setenv("ISSUE140_STARTUP_CAP","normal");m:=&Module{Threads:&ThreadPool{}};RegisterStartupCap(m,"bad");arg:=int32(1);phase:="load";switch bad{case "mode":t.Setenv("ISSUE140_STARTUP_CAP","bad");case "request":arg=4;case "phase":phase="other"};before:=len(startupEvents);defer func(){if recover()==nil{t.Fatal("invalid accepted")};if len(startupEvents)!=before||len(startupPending)!=0{t.Fatal("partial change")}}();StartupCapArg(m,128,arg,phase)})}
}
