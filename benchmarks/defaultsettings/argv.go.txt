package base

import (
	"encoding/binary"
	"fmt"
	"os"
	"strings"
)

var settingsVariant = os.Getenv("DEFAULT_SETTINGS_VARIANT")
var expectedSettingsArgs = []string{"lite4mariadb", "--basedir=/mariadb", "--datadir=/mariadb/data", "--plugin-dir=/mariadb/plugin", "--lc-messages-dir=/mariadb/share", "--character-sets-dir=/mariadb/share/charsets", "--tmpdir=/tmp", "--skip-grant-tables", "--skip-networking", "--skip-log-bin", "--innodb-use-native-aio=0", "--innodb-buffer-pool-size=16M", "--loose-innodb-buffer-pool-size-max=16M", "--innodb-log-buffer-size=2M", "--max-connections=10", "--default-storage-engine=InnoDB", "--loose-innodb-read-io-threads=1", "--loose-innodb-write-io-threads=1", "--loose-innodb-purge-threads=1", "--table-open-cache=32", "--thread-cache-size=0", "--socket=/tmp/mysql.sock", "--caching-sha2-password-private-key-path=/mariamem-auth/private.pem", "--caching-sha2-password-public-key-path=/mariamem-auth/public.pem", "--caching-sha2-password-auto-generate-rsa-keys=OFF"}
var removedSettingsArgs = map[string]bool{"max-connections": true, "loose-innodb-read-io-threads": true, "loose-innodb-write-io-threads": true, "loose-innodb-purge-threads": true, "table-open-cache": true, "thread-cache-size": true}

func SettingsArgv(m *Module, argc, argv int32) int32 {
	if settingsVariant != "current" && settingsVariant != "bounded-default" {
		panic("invalid DEFAULT_SETTINGS_VARIANT")
	}
	if argc != 25 {
		panic("unexpected original argc")
	}
	pointers := make([]uint32, argc)
	for i := range pointers {
		ea := MemoryEA(m, uint64(uint32(argv)), uint64(i*4), 4)
		p := binary.LittleEndian.Uint32(m.M[ea : ea+4])
		pointers[i] = p
		var b []byte
		for j := uint64(0); j < 512; j++ {
			v := m.M[MemoryEA(m, uint64(p), j, 1)]
			if v == 0 {
				break
			}
			b = append(b, v)
		}
		if string(b) != expectedSettingsArgs[i] {
			panic(fmt.Sprintf("argv[%d] preimage mismatch", i))
		}
	}
	count := 0
	for i, p := range pointers {
		if settingsVariant == "bounded-default" && removedSettingsArgs[strings.SplitN(strings.TrimPrefix(expectedSettingsArgs[i], "--"), "=", 2)[0]] {
			continue
		}
		ea := MemoryEA(m, uint64(uint32(argv)), uint64(count*4), 4)
		binary.LittleEndian.PutUint32(m.M[ea:ea+4], p)
		count++
	}
	ea := MemoryEA(m, uint64(uint32(argv)), uint64(count*4), 4)
	binary.LittleEndian.PutUint32(m.M[ea:ea+4], 0)
	if count != 25 && count != 19 {
		panic("unexpected filtered argc")
	}
	return int32(count)
}
