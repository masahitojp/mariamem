package base

import (
 "crypto/sha256"
 "encoding/hex"
 "fmt"
 "sync"
 "time"
 "unsafe"
)

// This diagnostic runs one guest at a time. Mode is set before Start;
// every initialization drains all records. No module/pointer is retained.
var LexerMode int
var lexerMu sync.Mutex
var lexerRows []LexerRow

type LexerRow struct {
 Number uint32 `json:"number"`
 Descriptor uint32 `json:"descriptor_offset"`
 ElapsedNS int64 `json:"elapsed_ns"`
 Result int32 `json:"result"`
 Exception int32 `json:"exception"`
 Pair string `json:"pair_hex,omitempty"`
 SHA256 string `json:"pair_sha256,omitempty"`
}

func LexerNow() time.Time {return time.Now()}
func LexerElapsed(t time.Time) int64 {return time.Since(t).Nanoseconds()}
func LexerU32(m *Module, p uint32) uint32 {
 return *(*uint32)(unsafe.Add(m.M,MemoryEA(m,uint64(p),0,4)))
}
func LexerObserve(m *Module, descriptor int32, result int32, ns int64) {
 row:=LexerRow{Number:LexerU32(m,uint32(descriptor)),Descriptor:uint32(descriptor),ElapsedNS:ns,Result:result,Exception:m.ExcPending}
 if LexerMode==2 && result==0 && m.ExcPending==0 {
  a:=LexerU32(m,uint32(descriptor)+72)
  z:=LexerU32(m,uint32(descriptor)+76)
  if a==0 || z!=a+256 {panic("lexer map layout mismatch")}
  bytes:=unsafe.Slice((*byte)(unsafe.Add(m.M,MemoryEA(m,uint64(a),0,512))),512)
  if err:=LexerValidatePair(bytes);err!=nil {panic(err)}
  sum:=sha256.Sum256(bytes);row.SHA256=hex.EncodeToString(sum[:]);row.Pair=hex.EncodeToString(bytes)
 }
 lexerMu.Lock();defer lexerMu.Unlock()
 if len(lexerRows)>=2048 {panic("lexer observation bound exceeded")}
 lexerRows=append(lexerRows,row)
}
func LexerDrain() []LexerRow {
 lexerMu.Lock();defer lexerMu.Unlock()
 r:=lexerRows;lexerRows=nil;return r
}
func LexerValidatePair(b []byte) error {
 if len(b)!=512 {return fmt.Errorf("map length %d",len(b))}
 specials:=map[byte]byte{0:18,'_':2,'$':2,'\'':10,'.':16,'>':8,'=':8,'!':8,'<':9,'&':17,'|':17,'#':11,';':22,':':23,'\\':19,'/':20,'*':21,'@':24,'`':27,'"':33,'-':34,',':36,'?':35,'x':30,'X':30,'b':31,'B':31,'n':32,'N':32}
 for k,v:=range specials {if b[k]!=v{return fmt.Errorf("state[%d]=%d want%d",k,b[k],v)}}
 for i:=0;i<256;i++ {
  want:=byte(0)
  // ident_map is created BEFORE the x/b/n overrides.
  if b[i]==2 || b[i]==14 || b[i]==30 || b[i]==31 || b[i]==32 {want=1}
  if b[256+i]!=want{return fmt.Errorf("ident[%d]=%d want%d",i,b[256+i],want)}
 }
 return nil
}
