import("context";"errors";"sync";"testing";"time")
type notifyFake struct {fakeBackend; mu sync.Mutex;n int;changed,failed,done chan struct{};cause error;registered chan struct{};afterSnapshot func()}
func newNotifyFake(n int)*notifyFake{return &notifyFake{n:n,changed:make(chan struct{}),failed:make(chan struct{}),done:make(chan struct{}),registered:make(chan struct{},64)}}
func(f *notifyFake)Active()int{f.mu.Lock();defer f.mu.Unlock();return f.n}
func(f *notifyFake)Failure()error{f.mu.Lock();defer f.mu.Unlock();return f.cause}
func(f *notifyFake)DisconnectState()(int,<-chan struct{},<-chan struct{},<-chan struct{}){f.mu.Lock();n,ch,failed,done:=f.n,f.changed,f.failed,f.done;hook:=f.afterSnapshot;f.afterSnapshot=nil;f.mu.Unlock();f.registered<-struct{}{};if hook!=nil{hook()};return n,ch,failed,done}
func(f *notifyFake)set(n int){f.mu.Lock();defer f.mu.Unlock();f.n=n;close(f.changed);f.changed=make(chan struct{})}
func(f *notifyFake)Close(context.Context)error{f.set(f.Active());return nil}
func notifyResult(t *testing.T,ch <-chan error)error{t.Helper();select{case e:=<-ch:return e;case <-time.After(time.Second):t.Fatal("wait stuck");return nil}}
func notifyRegistered(t *testing.T,f *notifyFake){t.Helper();select{case <-f.registered:case <-time.After(time.Second):t.Fatal("no state registration")}}
