	"database/sql"

	"github.com/shibukawa/pgmem/bench/modelcase/period"

)

type heldResult struct {
	DBs           int
	Period, Armed int64
	Readings      []period.Reading
	Expected      []int64
	After         [][]int64
	IdleMS        int64
	LowWrites     uint64
	Wakes         []wakeResult
}
type wakeResult struct {
	ID                        int
	Write                     bool
	Before, Connected         period.State
	ConnectMS, SQLMS, TotalMS float64
}

const readStock = "SELECT stock FROM products WHERE id=1"

func wakeDB(ctx context.Context, d *mm.Database, id int, write bool, stock int) (wakeResult, error) {
	r := wakeResult{ID: id, Write: write, Before: period.Inspect(id)}
	begin := time.Now()
	handle := openInProcess(d)
	defer handle.Close()
	conn, err := handle.Conn(ctx)
	if err != nil {
		return r, err
	}
	defer conn.Close()
	connected := time.Now()
	r.ConnectMS = float64(connected.Sub(begin)) / 1e6
	r.Connected = period.Inspect(id)
	sqlStart := time.Now()
	if write {
		result, err := conn.ExecContext(ctx, "UPDATE products SET stock=stock+1 WHERE id=1")
		if err != nil {
			return r, err
		}
		rows, err := result.RowsAffected()
		if err != nil || rows != 1 {
			return r, fmt.Errorf("update affected %d rows: %v", rows, err)
		}
	} else {
		var value int
		if err := conn.QueryRowContext(ctx, readStock).Scan(&value); err != nil {
			return r, err
		}
		if value != stock {
			return r, fmt.Errorf("stock %d, want %d", value, stock)
		}
	}
	end := time.Now()
	r.SQLMS = float64(end.Sub(sqlStart)) / 1e6
	r.TotalMS = float64(end.Sub(begin)) / 1e6
	return r, nil
}
func orderTotals(ctx context.Context, db *sql.DB) ([]int64, error) {
	values := make([]int64, 4)
	err := db.QueryRowContext(ctx, "SELECT (SELECT COUNT(*) FROM orders), (SELECT COUNT(*) FROM order_items), (SELECT SUM(total_cents) FROM orders), (SELECT SUM(quantity*unit_price_cents) FROM order_items)").Scan(&values[0], &values[1], &values[2], &values[3])
	return values, err
}
func holdIdle() error {
	n, err := strconv.Atoi(os.Getenv("IDLE_DBS"))
	if err != nil || n != 1 && n != 8 {
		return fmt.Errorf("invalid idle size")
	}
	ctx := context.Background()
	root, err := mm.Start(ctx, mm.Options{})
	if err != nil {
		return err
	}
	sqlRoot := openInProcess(root)
	if err = prepare(ctx, sqlRoot); err != nil {
		return err
	}
	if err = sqlRoot.Close(); err != nil {
		return err
	}
	if err = root.WaitDisconnected(ctx); err != nil {
		return err
	}
	snap, err := root.Snapshot(ctx, mm.SnapshotOptions{MaxForks: 16})
	if err != nil {
		return err
	}
	if err = root.Close(); err != nil {
		return err
	}
	var dbs []*mm.Database
	var ids, stocks []int
	var total int64
	for _, order := range store.SampleOrders() {
		for _, it := range order.Items {
			total += int64(it.Quantity) * int64(it.UnitPriceCents)
		}
	}
	result := heldResult{DBs: n, Period: period.Milliseconds, Expected: []int64{40, 100, total, total}}
	check := func(s *mm.Database, load bool) ([]int64, error) {
		d := openInProcess(s)
		if load {
			if err := store.LoadSample(ctx, d); err != nil {
				return nil, err
			}
		}
		v, err := orderTotals(ctx, d)
		if err != nil {
			return nil, err
		}
		for i, x := range v {
			if result.Expected[i] != x {
				return nil, fmt.Errorf("order checks %v; expected %v", v, result.Expected)
			}
		}
		var stock int
		if err = d.QueryRowContext(ctx, readStock).Scan(&stock); err != nil {
			return nil, err
		}
		if load {
			stocks = append(stocks, stock)
		} else {
			index := len(result.After)
			want := stocks[index]
			if result.Wakes[index].Write {
				want++
			}
			if stock != want {
				return nil, fmt.Errorf("stock after wake %d, want %d", stock, want)
			}
		}
		if err = d.Close(); err != nil {
			return nil, err
		}
		return v, s.WaitDisconnected(ctx)
	}
	for i := 0; i < n; i++ {
		s, err := snap.Fork(ctx)
		if err != nil {
			return err
		}
		ids = append(ids, period.LatestID())
		if ids[i] != i+2 {
			return fmt.Errorf("unexpected pool identity %v", ids)
		}
		dbs = append(dbs, s)
		if _, err = check(s, true); err != nil {
			return err
		}
	}
	if err = snap.Close(); err != nil {
		return err
	}
	result.Armed = period.ArmCount()
	if result.Armed != int64(n+1) {
		return fmt.Errorf("active DBs %d, arms %d", n, result.Armed)
	}
	origin := time.Now()
	stops := []int{0, 30, 70, 120}
	if os.Getenv("IDLE_PILOT") == "1" {
		stops = []int{0, 1, 2, 3}
	}
	result.Readings = make([]period.Reading, 0, len(stops))
	for _, sec := range stops {
		time.Sleep(time.Until(origin.Add(time.Duration(sec) * time.Second)))
		result.Readings = append(result.Readings, period.Read(origin))
	}
	result.IdleMS = period.IdleMS()
	result.LowWrites = period.LowWrites()
	reverse := os.Getenv("IDLE_REVERSE") == "1"
	for i, d := range dbs {
		deadline := origin.Add(time.Duration(stops[len(stops)-1])*time.Second + time.Duration(i)*450*time.Millisecond)
		time.Sleep(time.Until(deadline))
		wake, err := wakeDB(ctx, d, ids[i], (i%2 == 1) != reverse, stocks[i])
		if err != nil {
			return err
		}
		result.Wakes = append(result.Wakes, wake)
		if err = d.WaitDisconnected(ctx); err != nil {
			return err
		}
	}
	for _, d := range dbs {
		v, err := check(d, false)
		if err != nil {
			return err
		}
		result.After = append(result.After, v)
		if err = d.Close(); err != nil {
			return err
		}
	}
	encoded, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(os.Getenv("IDLE_REPORT"), encoded, 0644)
}
