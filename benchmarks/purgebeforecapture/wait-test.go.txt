package probe
import("context";"errors";"testing";"time")
func TestPurgePolicyBoundsAndTarget(t *testing.T){for _,tc:=range []struct{name,mode string;values []uint64;wantMS float64;target,timeout bool;reads int}{{"now","now",[]uint64{5},0,false,false,1},{"progress","drain",[]uint64{5,3,0},100,true,false,6},{"already_zero","drain",[]uint64{0},100,true,false,6},{"stuck","drain",[]uint64{2},1000,false,true,51}}{t.Run(tc.name,func(t *testing.T){at:=time.Unix(0,0);i:=0;out,e:=purgePolicy(context.Background(),tc.mode,func()time.Time{return at},func(ctx context.Context,d time.Duration)error{if d<=0||d>20*time.Millisecond{t.Fatal(d)};at=at.Add(d);return nil},func(context.Context)(uint64,error){idx:=i;if idx>=len(tc.values){idx=len(tc.values)-1};i++;return tc.values[idx],nil});if e!=nil||out.ElapsedMS!=tc.wantMS||out.TargetReached!=tc.target||out.TimedOut!=tc.timeout||len(out.Points)!=tc.reads{t.Fatal(out,e)}})}}
func TestPurgePolicyErrors(t *testing.T){sentinel:=errors.New("read");for _,mode:=range []string{"now","drain"}{out,e:=purgePolicy(context.Background(),mode,time.Now,func(context.Context,time.Duration)error{return nil},func(context.Context)(uint64,error){return 0,sentinel});if !errors.Is(e,sentinel)||len(out.Points)!=0{t.Fatal(out,e)}}
 out,e:=purgePolicy(context.Background(),"invalid",time.Now,func(context.Context,time.Duration)error{return nil},func(context.Context)(uint64,error){t.Fatal("unexpected read");return 0,nil});if e==nil||len(out.Points)!=0{t.Fatal(out,e)}
 out,e=purgePolicy(context.Background(),"drain",time.Now,func(context.Context,time.Duration)error{return context.Canceled},func(context.Context)(uint64,error){return 1,nil});if !errors.Is(e,context.Canceled)||out.TimedOut{t.Fatal(out,e)}
}
