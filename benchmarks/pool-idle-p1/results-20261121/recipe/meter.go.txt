import "C"
import (
    "sync/atomic"
    "syscall"
    "time"
)

var callbacks, low, high, workers atomic.Uint64
func Tick() { callbacks.Add(1) }
func Low() { low.Add(1) }
func High() { high.Add(1) }
func StartWorker() { workers.Add(1) }
type Reading struct {
    Seconds float64
    UserUS, SystemUS int64
    Interrupts, PackageIdle, Ticks, SetLow, SetHigh, Starts uint64
    Voluntary, Involuntary int64
}
func Read(origin time.Time) Reading {
    var u syscall.Rusage
    if err:=syscall.Getrusage(syscall.RUSAGE_SELF,&u);err!=nil{panic(err)}
    var irq, pkg C.uint64_t
    if rc:=C.wake_counts(&irq,&pkg);rc!=0{panic("proc_pid_rusage failed")}
    return Reading{time.Since(origin).Seconds(),u.Utime.Sec*1e6+int64(u.Utime.Usec),u.Stime.Sec*1e6+int64(u.Stime.Usec),
        uint64(irq),uint64(pkg),callbacks.Load(),low.Load(),high.Load(),workers.Load(),u.Nvcsw,u.Nivcsw}
}
