type g2Oracle struct {
	Additional []struct {
		Name  string
		Type  int
		Value string
		Trace []map[string]any
	}
	Long []struct {
		Name  string
		Trace [][]map[string]any
	}
	Prepare []struct {
		Name, SQL string
		Expected  map[string]any
	}
	Values []struct {
		Name, Packet string
		Expected     map[string]any
	}
	Calls []struct {
		Name string
		Spec struct {
			Args, Body     string
			Values, Second []*int
			PS             *bool
			Alter          bool
		}
		Trace [][]map[string]any
	}
}

func g2Gold(t *testing.T) g2Oracle {
	t.Helper()
	var o g2Oracle
	if e := json.Unmarshal([]byte(g2OracleJSON), &o); e != nil {
		t.Fatal(e)
	}
	return o
}
func g2Observe(t *testing.T, id string, want, actual any, pass bool) {
	t.Helper()
	b, _ := json.Marshal(map[string]any{"test": t.Name(), "id": id, "expected": want, "actual": actual, "pass": pass})
	t.Log("G2_OBSERVATION " + string(b))
	if !pass {
		t.Errorf("G2 contract mismatch: %s", id)
	}
}
func g2Normalize(x any) any {
	b, _ := json.Marshal(x)
	var out any
	json.Unmarshal(b, &out)
	return out
}
func g2Equal(a, b any) bool { return reflect.DeepEqual(g2Normalize(a), g2Normalize(b)) }
func g2Hex(p []byte) string { return hex.EncodeToString(p) }
func g2Error(p []byte) map[string]any {
	n, s := compatErr(p)
	return map[string]any{"error_packet": g2Hex(p), "errno": n, "sqlstate": s}
}
func g2Len(p []byte, at *int) (uint64, error) {
	if *at >= len(p) {
		return 0, io.ErrUnexpectedEOF
	}
	v := p[*at]
	*at = *at + 1
	switch v {
	case 252:
		if len(p)-*at < 2 {
			return 0, io.ErrUnexpectedEOF
		}
		n := binary.LittleEndian.Uint16(p[*at:])
		*at += 2
		return uint64(n), nil
	case 253:
		if len(p)-*at < 3 {
			return 0, io.ErrUnexpectedEOF
		}
		n := uint64(p[*at]) | uint64(p[*at+1])<<8 | uint64(p[*at+2])<<16
		*at += 3
		return n, nil
	case 254:
		if len(p)-*at < 8 {
			return 0, io.ErrUnexpectedEOF
		}
		n := binary.LittleEndian.Uint64(p[*at:])
		*at += 8
		return n, nil
	case 251, 255:
		return 0, fmt.Errorf("invalid integer marker %d", v)
	}
	return uint64(v), nil
}

// Finite fixture reader:160 results,128 columns,2000 rows. These are test
// workload bounds,not product ceilings; the native129-result oracle is included.
func g2Packet(c *compatConn, seq byte) ([]byte, error) {
	c.c.SetReadDeadline(time.Now().Add(12 * time.Second))
	var h [4]byte
	if _, e := io.ReadFull(c.c, h[:]); e != nil {
		return nil, e
	}
	n := int(h[0]) | int(h[1])<<8 | int(h[2])<<16
	if n == 0 || n > 0xfffffe || h[3] != seq {
		return nil, fmt.Errorf("packet header len%d seq%d want%d", n, h[3], seq)
	}
	p := make([]byte, n)
	_, e := io.ReadFull(c.c, p)
	return p, e
}
func g2Read(c *compatConn) ([]map[string]any, error) {
	seq := byte(1)
	read := func() ([]byte, error) { p, e := g2Packet(c, seq); seq++; return p, e }
	var parts []map[string]any
	for len(parts) < 160 {
		p, e := read()
		if e != nil {
			return parts, e
		}
		if p[0] == 255 {
			return append(parts, g2Error(p)), nil
		}
		var part map[string]any
		var status uint16
		if p[0] == 0 {
			at := 1
			if _, e = g2Len(p, &at); e != nil {
				return parts, e
			}
			if _, e = g2Len(p, &at); e != nil {
				return parts, e
			}
			if len(p)-at < 4 {
				return parts, io.ErrUnexpectedEOF
			}
			status = binary.LittleEndian.Uint16(p[at:])
			part = map[string]any{"ok_packet": g2Hex(p), "status": status}
		} else {
			at := 0
			count, e := g2Len(p, &at)
			if e != nil || count == 0 || count > 128 {
				return parts, fmt.Errorf("column count %d: %v", count, e)
			}
			cols := []string{}
			for i := uint64(0); i < count; i++ {
				p, e = read()
				if e != nil {
					return parts, e
				}
				cols = append(cols, g2Hex(p))
			}
			meta, e := read()
			if e != nil {
				return parts, e
			}
			if len(meta) != 5 || meta[0] != 254 {
				return parts, fmt.Errorf("metadata EOF %x", meta)
			}
			rows := []string{}
			for {
				p, e = read()
				if e != nil {
					return parts, e
				}
				if p[0] == 255 {
					part = g2Error(p)
					part["columns"] = cols
					part["rows"] = rows
					part["metadata_end"] = g2Hex(meta)
					return append(parts, part), nil
				}
				if p[0] == 254 && len(p) < 9 {
					if len(p) != 5 {
						return parts, fmt.Errorf("terminal EOF %x", p)
					}
					break
				}
				if len(rows) >= 2000 {
					return parts, fmt.Errorf("fixture row bound")
				}
				rows = append(rows, g2Hex(p))
			}
			status = binary.LittleEndian.Uint16(p[3:])
			part = map[string]any{"columns": cols, "rows": rows, "metadata_end": g2Hex(meta), "terminal": g2Hex(p), "status": status}
		}
		parts = append(parts, part)
		if status&8 == 0 {
			return parts, nil
		}
	}
	return parts, fmt.Errorf("fixture result bound")
}
func g2Command(c *compatConn, p []byte) ([]map[string]any, error) {
	if e := c.send(0, p); e != nil {
		return nil, e
	}
	return g2Read(c)
}
func g2Setup(t *testing.T, c *compatConn, sql string) {
	t.Helper()
	p, e := g2Command(c, append([]byte{3}, sql...))
	if e != nil || len(p) != 1 || p[0]["ok_packet"] == nil {
		t.Fatalf("setup %q: %v %v", sql, p, e)
	}
}
func g2Login(t *testing.T, c *compatConn, ps bool, db string, charset byte) ([]byte, error) {
	flags := uint32(1 | 4 | 8 | 512 | 8192 | 32768 | 131072)
	if ps {
		flags |= 262144
	}
	return g2LoginFlags(t, c, flags, db, charset)
}
func g2LoginFlags(t *testing.T, c *compatConn, flags uint32, db string, charset byte) ([]byte, error) {
	t.Helper()
	c.greeting = c.mustRead(t, 0)
	p := make([]byte, 32)
	binary.LittleEndian.PutUint32(p, flags)
	p[8] = charset
	p = append(p, []byte("root\x00\x00")...)
	p = append(p, db...)
	p = append(p, 0)
	if e := c.send(1, p); e != nil {
		return nil, e
	}
	return c.read(2)
}
func g2Connect(t *testing.T, address string, ps bool) *compatConn {
	t.Helper()
	c := compatConnect(t, address, nil)
	p, e := g2Login(t, c, ps, "test", 45)
	if e != nil || len(p) == 0 || p[0] != 0 {
		t.Fatalf("login %x %v", p, e)
	}
	g2Setup(t, c, "SET NAMES utf8mb4 COLLATE utf8mb4_general_ci")
	g2Setup(t, c, "SET sql_mode='', time_zone='+00:00'")
	return c
}
func g2Prepare(c *compatConn, sql string) (uint32, map[string]any, error) {
	if e := c.send(0, append([]byte{22}, sql...)); e != nil {
		return 0, nil, e
	}
	p, e := g2Packet(c, 1)
	if e != nil {
		return 0, nil, e
	}
	if p[0] == 255 {
		return 0, g2Error(p), nil
	}
	if len(p) != 12 || p[0] != 0 {
		return 0, nil, fmt.Errorf("prepare %x", p)
	}
	id := binary.LittleEndian.Uint32(p[1:])
	seq := byte(2)
	groups := [][]string{{}, {}}
	for i, n := range []uint16{binary.LittleEndian.Uint16(p[7:]), binary.LittleEndian.Uint16(p[5:])} {
		if n > 128 {
			return id, nil, fmt.Errorf("fixture prepare count")
		}
		for j := uint16(0); j < n; j++ {
			q, e := g2Packet(c, seq)
			seq++
			if e != nil {
				return id, nil, e
			}
			groups[i] = append(groups[i], g2Hex(q))
		}
		if n > 0 {
			q, e := g2Packet(c, seq)
			seq++
			if e != nil || len(q) != 5 || q[0] != 254 {
				return id, nil, fmt.Errorf("prepare EOF %x %v", q, e)
			}
		}
	}
	return id, map[string]any{"parameters": groups[0], "columns": groups[1], "warnings": binary.LittleEndian.Uint16(p[10:])}, nil
}
func g2Following(t *testing.T, c *compatConn, name string) {
	t.Helper()
	p, e := g2Command(c, append([]byte{3}, "SELECT 1, @p3_after_error"...))
	want := []string{"0131fb"}
	ok := e == nil && len(p) == 1 && g2Equal(p[0]["rows"], want)
	g2Observe(t, name, "following SELECT 1,NULL", map[string]any{"parts": p, "error": fmt.Sprint(e)}, ok)
}

}

		})
	}
}
func g2Ints(id uint32, values []*int, reuse bool) []byte {
	if len(values) == 0 {
		return g1Exec(id, 0, 1, nil)
	}
	p := make([]byte, (len(values)+7)/8)
	for i, v := range values {
		if v == nil {
			p[i/8] |= 1 << uint(i%8)
		}
	}
	if reuse {
		p = append(p, 0)
	} else {
		p = append(p, 1)
		for range values {
			p = append(p, 3, 0)
		}
	}
	for _, v := range values {
		if v != nil {
			var b [4]byte
			binary.LittleEndian.PutUint32(b[:], uint32(*v))
			p = append(p, b[:]...)
		}
	}
	return g1Exec(id, 0, 1, p)
}
func TestG2CallChains(t *testing.T) {
	for _, tc := range g2Gold(t).Calls {
		t.Run(tc.Name, func(t *testing.T) {
			r := rootFor(t, 10000)
			ps := tc.Spec.PS == nil || *tc.Spec.PS
			c := g2Connect(t, compatAddress(t, r), ps)
			g2Setup(t, c, "SET @p3_after_error=0, @p3_marker=0")
			if tc.Spec.Alter {
				g2Setup(t, c, "CREATE TABLE test.prepared_p3_matrix_items(id INT PRIMARY KEY,v VARCHAR(12)) ENGINE=InnoDB")
				g2Setup(t, c, "INSERT INTO test.prepared_p3_matrix_items VALUES(1,'seed')")
			}
			g2Setup(t, c, "CREATE PROCEDURE test.prepared_p3_matrix_proc("+tc.Spec.Args+") BEGIN "+tc.Spec.Body+" END")
			id, meta, e := g2Prepare(c, "CALL test.prepared_p3_matrix_proc("+strings.TrimSuffix(strings.Repeat("?,", len(tc.Spec.Values)), ",")+")")
			if e != nil || id == 0 {
				t.Fatal("prepare", meta, e)
			}
			for i, want := range tc.Trace {
				values := tc.Spec.Values
				if i > 0 {
					if tc.Spec.Alter {
						g2Setup(t, c, "ALTER TABLE test.prepared_p3_matrix_items MODIFY v VARBINARY(12), ADD added BIGINT DEFAULT 9")
					} else {
						values = tc.Spec.Second
					}
				}
				parts, e := g2Command(c, g2Ints(id, values, i > 0))
				g2Observe(t, fmt.Sprintf("%s_trace%d", tc.Name, i), want, map[string]any{"parts": parts, "error": fmt.Sprint(e)}, e == nil && g2Equal(parts, want))
				following, e := g2Command(c, append([]byte{3}, "SELECT 1, @p3_after_error"...))
				g2Observe(t, fmt.Sprintf("%s_following%d", tc.Name, i), "row1,0 after complete chain", map[string]any{"parts": following, "error": fmt.Sprint(e)}, e == nil && len(following) == 1 && g2Equal(following[0]["rows"], []string{"01310130"}))
			}
			c.mustSend(t, 0, compatID(25, id))
			g2Ping(t, c, tc.Name+"_close_ping")
		})
	}
}
func g2Value(id uint32, typ byte, value []byte, reuse bool) []byte {
	p := []byte{0, 0}
	if !reuse {
		p = []byte{0, 1, typ, 0}
	}
	switch typ {
	case 1, 2, 3, 4, 5, 8, 9, 13:
		p = append(p, value...)
	case 7, 10, 11, 12, 14:
		p = append(p, byte(len(value)))
		p = append(p, value...)
	default:
		p = append(p, byte(len(value)))
		p = append(p, value...)
	}
	return g1Exec(id, 0, 1, p)
}

func g2Chunk(t *testing.T, c *compatConn, id uint32, index uint16, data []byte) {
	t.Helper()
	p := compatID(24, id)
	p = append(p, byte(index), byte(index>>8))
	c.mustSend(t, 0, append(p, data...))
}
func g2Stream(id uint32, types []byte, mask byte, reuse bool, inline []byte) []byte {
	p := []byte{mask, 0}
	if !reuse {
		p[1] = 1
		for _, typ := range types {
			p = append(p, typ, 0)
		}
	}
	return g1Exec(id, 0, 1, append(p, inline...))
}
func TestG2LongDataStates(t *testing.T) {
	for _, tc := range g2Gold(t).Long {
		t.Run(tc.Name, func(t *testing.T) {
			r := rootFor(t, 10000)
			c := g2Connect(t, compatAddress(t, r), true)
			trace := [][]map[string]any{}
			errs := []string{}
			execute := func(p []byte) {
				parts, e := g2Command(c, p)
				trace = append(trace, parts)
				errs = append(errs, fmt.Sprint(e))
			}
			switch tc.Name {
			case "chunks-binary-mixed-parameters":
				id := c.prepare(t, "SELECT ? AS a,? AS n,? AS b")
				g2Chunk(t, c, id, 0, []byte{'A', 0})
				g2Chunk(t, c, id, 2, []byte("日本語"))
				g2Chunk(t, c, id, 0, []byte("B"))
				execute(g2Stream(id, []byte{252, 3, 252}, 4, false, []byte{23, 0, 0, 0}))
			case "empty-stream-overrides-null":
				id := c.prepare(t, "SELECT ? AS v")
				g2Chunk(t, c, id, 0, nil)
				g2Chunk(t, c, id, 0, nil)
				execute(g2Stream(id, []byte{252}, 1, false, nil))
				execute(g1Exec(id, 0, 1, []byte{1, 1, 252, 0}))
			case "type-reuse-and-change":
				id := c.prepare(t, "SELECT ? AS v")
				execute(g2Value(id, 252, []byte("first"), false))
				g2Chunk(t, c, id, 0, []byte("long"))
				g2Chunk(t, c, id, 0, []byte("-reuse"))
				execute(g2Stream(id, []byte{252}, 0, true, nil))
				execute(g2Value(id, 252, []byte("inline"), true))
				g2Chunk(t, c, id, 0, []byte("文字"))
				execute(g2Stream(id, []byte{253}, 0, false, nil))
			case "two-statement-ownership":
				a := c.prepare(t, "SELECT ? AS v")
				b := c.prepare(t, "SELECT ? AS v")
				g2Chunk(t, c, a, 0, []byte("first"))
				g2Chunk(t, c, b, 0, []byte("second"))
				execute(g2Stream(b, []byte{252}, 0, false, nil))
				execute(g2Stream(a, []byte{252}, 0, false, nil))
			case "wrong-LONG-type":
				id := c.prepare(t, "SELECT ? AS v")
				g2Chunk(t, c, id, 0, []byte("abc"))
				execute(g2Stream(id, []byte{3}, 0, false, nil))
				execute(g2Value(id, 252, []byte("z"), false))
			case "reset-stream":
				id := c.prepare(t, "SELECT ? AS v")
				execute(g2Value(id, 252, []byte("first"), false))
				g2Chunk(t, c, id, 0, []byte("discarded"))
				reply := g1Send(c, compatID(26, id), false, false)
				g2Observe(t, tc.Name+"_reset", "RESET OK", reply, reply.Kind == "OK" && reply.Error == "")
				execute(g2Value(id, 252, []byte("new"), true))
			}
			ok := g2Equal(trace, tc.Trace)
			for _, e := range errs {
				ok = ok && e == "<nil>"
			}
			g2Observe(t, tc.Name, tc.Trace, map[string]any{"trace": trace, "errors": errs}, ok)
			g2Ping(t, c, tc.Name+"_following")
		})
	}
	t.Run("malformed_preserves_stream", func(t *testing.T) {
		r := rootFor(t, 10000)
		c := g2Connect(t, compatAddress(t, r), true)
		id := c.prepare(t, "SELECT CAST(? AS CHAR)")
		seed := g1Send(c, g2Value(id, 252, []byte("seed"), false), true, false)
		if seed.Kind != "ROW" || seed.Value != "seed" {
			t.Fatal(seed)
		}
		g2Chunk(t, c, id, 0, []byte("kept"))
		bad := g1Send(c, g1Exec(id, 0, 1, []byte{0, 1, 252, 1}), true, false)
		g2Observe(t, "malformed_rejected", "1835/HY000", bad, bad.Kind == "ERR" && bad.Errno == 1835 && bad.State == "HY000")
		got := g1Send(c, g1Exec(id, 0, 1, []byte{0, 0}), true, false)
		g2Observe(t, "malformed_retains_stream", "pending LONG_DATA kept after rejected descriptor", got, got.Kind == "ROW" && got.Value == "kept" && got.Error == "")
		fresh := g1Send(c, g2Value(id, 252, []byte("fresh"), false), true, false)
		g2Observe(t, "malformed_fresh", "fresh recovery", fresh, fresh.Kind == "ROW" && fresh.Value == "fresh" && fresh.Error == "")
		g2Ping(t, c, "malformed_following")
	})
}
func TestG2LifecycleBodies(t *testing.T) {
	for n := 0; n <= 5; n++ {
		t.Run(fmt.Sprint(n), func(t *testing.T) {
			r := rootFor(t, 10000)
			c := g2Connect(t, compatAddress(t, r), true)
			id := c.prepare(t, "SELECT ?")
			if id >= 256 {
				t.Fatal("fixture needs small id")
			}
			body := make([]byte, max(n, 4))
			binary.LittleEndian.PutUint32(body, id)
			if n == 5 {
				body[4] = 'x'
			}
			body = body[:n]
			reply := g1Send(c, append([]byte{26}, body...), false, false)
			valid := reply.Error == "" && reply.Kind == "OK"
			want := "RESET OK"
			if n == 0 {

func g2FieldName(encoded string) (string, int, error) {
	p, e := hex.DecodeString(encoded)
	if e != nil {
		return "", 0, e
	}
	at := 0
	name := ""
	for i := 0; i < 6; i++ {
		n, e := g2Len(p, &at)
		if e != nil || n > uint64(len(p)-at) {
			return "", len(p), fmt.Errorf("descriptor string %d", i)
		}
		if i == 4 {
			name = string(p[at : at+int(n)])
		}
		at += int(n)
	}
	if len(p)-at != 13 || p[at] != 12 {
		return "", len(p), fmt.Errorf("descriptor fixed suffix")
	}
	return name, len(p), nil
}

func g2WaitCause(t *testing.T, r *instance) {
	t.Helper()
	deadline := time.Now().Add(5 * time.Second)
	for r.result() == nil && time.Now().Before(deadline) {
		time.Sleep(time.Millisecond)
	}
	if r.result() == nil {
		t.Fatal("fixture: owner deadline did not arrive")
	}
}

// Validate the remainder of the known40x1MiB text-row stream through EOF,
// allowing a truncated final native packet and rejecting an injected wire ERR.
func g2DrainRows(c *compatConn, firstLen int) (int64, error) {
	total := int64(0)
	remaining := firstLen - 4 - 16
	seq := byte(5)
	buf := make([]byte, 32768)
	for packets := 0; packets < 41; packets++ {
		for remaining > 0 {
			n, e := c.c.Read(buf[:min(len(buf), remaining)])
			for _, v := range buf[:n] {
				if v != 'x' {
					return total, fmt.Errorf("non-native row byte %02x", v)
				}
			}
			total += int64(n)
			remaining -= n
			if e != nil {
				if limitsClosed(e) {
					return total, nil
				}
				return total, e
			}
		}
		var h [4]byte
		if _, e := io.ReadFull(c.c, h[:]); e != nil {
			if limitsClosed(e) {
				return total, nil
			}
			return total, e
		}
		size := int(h[0]) | int(h[1])<<8 | int(h[2])<<16
		if h[3] != seq || size == 0 || size > 0xfffffe {
			return total, fmt.Errorf("frame %x expected%d", h, seq)
		}
		seq++
		if size == 5 {
			return total, fmt.Errorf("normal terminal arrived before fatal EOF")
		}
		if size != 1048580 {
			return total, fmt.Errorf("unexpected row size%d", size)
		}
		var prefix [4]byte
		if _, e := io.ReadFull(c.c, prefix[:]); e != nil {
			if limitsClosed(e) {
				return total, nil
			}
			return total, e
		}
		if prefix != [4]byte{253, 0, 0, 16} {
			return total, fmt.Errorf("row prefix %x", prefix)
		}
		remaining = size - 4
	}
	return total, fmt.Errorf("fixture row-count bound")
}
func TestG2FatalStates(t *testing.T) {
	for _, name := range []string{"partial_body", "mid_output", "buffered_future"} {
		t.Run(name, func(t *testing.T) {
			ms := uint64(1000)
			if name == "mid_output" {
				ms = 2000
			}
			bad := rootFor(t, ms)
			good := rootFor(t, 10000)
			address := compatAddress(t, bad)
			peer := g2Connect(t, address, true)
			active := g2Connect(t, address, true)
			nativePrefix := false
			rowSize := 0
			switch name {
			case "partial_body":
				active.c.SetWriteDeadline(time.Now().Add(time.Second))
				p := []byte{64, 0, 0, 0, 3, 'S', 'E'}
				n, e := active.c.Write(p)
				if e != nil || n != len(p) {
					t.Fatal("partial body", n, e)
				}
				nativePrefix = true
			case "buffered_future":
				active.mustSend(t, 0, append([]byte{3}, "SELECT SLEEP(4)"...))
				inspect(t, bad, func(s map[string]any) bool {
					for _, v := range s["sessions"].([]any) {
						if v.(map[string]any)["phase"] == "executing" {
							return true
						}
					}
					return false
				})
				active.mustSend(t, 0, []byte{14})
				active.mustSend(t, 0, append([]byte{3}, "SELECT 777"...))
				inspect(t, bad, func(s map[string]any) bool {
					for _, v := range s["sessions"].([]any) {
						if v.(map[string]any)["input"].(float64) >= 5 {
							return true
						}
					}
					return false
				})
				nativePrefix = true
			case "mid_output":
				active.mustSend(t, 0, append([]byte{3}, "SELECT REPEAT('x',1048576) FROM (SELECT 1 UNION ALL SELECT 2 UNION ALL SELECT 3 UNION ALL SELECT 4 UNION ALL SELECT 5 UNION ALL SELECT 6 UNION ALL SELECT 7 UNION ALL SELECT 8) a CROSS JOIN (SELECT 1 UNION ALL SELECT 2 UNION ALL SELECT 3 UNION ALL SELECT 4 UNION ALL SELECT 5) b"...))
				first := active.mustRead(t, 1)
				column := active.mustRead(t, 2)
				meta := active.mustRead(t, 3)
				var h [4]byte
				active.c.SetReadDeadline(time.Now().Add(3 * time.Second))
				if _, e := io.ReadFull(active.c, h[:]); e != nil {
					t.Fatal(e)
				}
				rowSize = int(h[0]) | int(h[1])<<8 | int(h[2])<<16
				var prefix [20]byte
				if _, e := io.ReadFull(active.c, prefix[:]); e != nil {
					t.Fatal(e)
				}
				nativePrefix = bytes.Equal(first, []byte{1}) && len(column) > 0 && len(meta) == 5 && meta[0] == 254 && rowSize == 1048580 && h[3] == 4 && bytes.Equal(prefix[:4], []byte{253, 0, 0, 16}) && bytes.Equal(prefix[4:], bytes.Repeat([]byte{'x'}, 16))
			}
			g2WaitCause(t, bad)
			var transportErr error
			var buffered int64
			if name == "mid_output" {
				active.c.SetReadDeadline(time.Now().Add(3 * time.Second))
				buffered, transportErr = g2DrainRows(active, rowSize)
			} else {
				p, e := active.read(1)
				transportErr = e
				if len(p) > 0 {
					transportErr = fmt.Errorf("unexpected following packet%x", p)
				}
			}
			eof := limitsClosed(transportErr)
			if name == "mid_output" {
				eof = transportErr == nil
			}
			g2Observe(t, name+"_active", "native prefix(if emitted) then EOF;no injected ERR/future reply", map[string]any{"prefix_valid": nativePrefix, "drained_native_bytes": buffered, "error": fmt.Sprint(transportErr)}, nativePrefix && eof)
			_, e := peer.read(1)
			g2Observe(t, name+"_peer", "same-instance idle peer EOF", fmt.Sprint(e), limitsClosed(e))
			cause := bad.result()
			g2Observe(t, name+"_owner", "owner original deadline cause", fmt.Sprint(cause), cause != nil && strings.Contains(cause.Error(), "deadline"))
			independent := g2Connect(t, compatAddress(t, good), true)
			g2Ping(t, independent, name+"_independent")
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			defer cancel()
			closeErr := bad.close()
			joined := bad.awaitJoined(ctx)
			g2Observe(t, name+"_cleanup", "joined and original deadline retained", map[string]any{"close": fmt.Sprint(closeErr), "join": fmt.Sprint(joined)}, closeErr != nil && strings.Contains(closeErr.Error(), "deadline") && joined != nil && strings.Contains(joined.Error(), "deadline"))
		})
	}
}
func g2TableAlias(p []byte) ([]byte, error) {
	at := 0
	for i := 0; i < 6; i++ {
		n, e := g2Len(p, &at)
		if e != nil || n > uint64(len(p)-at) {
			return nil, fmt.Errorf("metadata string%d", i)
		}
		if i == 2 {
			return p[at : at+int(n)], nil
		}
		at += int(n)
	}
	return nil, fmt.Errorf("missing table alias")
}

			return false, map[string]any{"sql": sql, "parts": p, "error": fmt.Sprint(e)}
		}
	}
	return true, "internal SQL prepared"
}

					if e == nil {
						end, e = c.read(5)
					}
				}

			}
			g2Ping(t, c, name+"_following")
		})
	}
