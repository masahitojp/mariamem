package main

import (
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"os"

	"github.com/goccy/wasm2go/internal/wasm"
)

func main() {
	for _, path := range os.Args[1:] {
		f, e := os.Open(path)
		if e != nil {
			panic(e)
		}
		m, e := wasm.Parse(f)
		f.Close()
		if e != nil {
			panic(e)
		}
		rows := []any{}
		for i, fn := range m.Functions {
			r := wasm.NewInstrReader(fn.Body)
			n, e := r.ReadU32()
			if e != nil {
				panic(e)
			}
			for j := uint32(0); j < n; j++ {
				if _, e := r.ReadU32(); e != nil {
					panic(e)
				}
				if _, e := r.ReadByte(); e != nil {
					panic(e)
				}
			}
			counts := map[string]int{}
			fences := []int{}
			for !r.EOF() {
				p := r.Pos()
				op, e := r.ReadByte()
				if e != nil {
					panic(e)
				}
				key := fmt.Sprintf("%02x", op)
				if op == 0xfe {
					peek := wasm.NewInstrReader(r.Src()[r.Pos():])
					sub, e := peek.ReadU32()
					if e != nil {
						panic(e)
					}
					key += fmt.Sprintf(".%02x", sub)
					if sub == 3 {
						fences = append(fences, p)
					}
				}
				counts[key]++
				if e := r.SkipImmediates(op); e != nil {
					panic(e)
				}
			}
			rows = append(rows, map[string]any{"defined_index": i, "body_sha256": fmt.Sprintf("%x", sha256.Sum256(fn.Body)), "opcodes": counts, "fences": fences})
		}
		if e := json.NewEncoder(os.Stdout).Encode(map[string]any{"path": path, "functions": rows}); e != nil {
			panic(e)
		}
	}
}
