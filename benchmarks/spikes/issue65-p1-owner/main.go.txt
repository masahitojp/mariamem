const (
	open       = 1
	data       = 2
	ack        = 3
	peerEOF    = 4
	quiesce    = 5
	stop       = 6
	stats      = 7
	ready      = 100
	opened     = 101
	raw        = 103
	retired    = 104
	poison     = 105
	credit     = 106
	state      = 107
	statsReply = 108
	inputLimit = 1048581
)

type frame struct {
	kind                     byte
	request, slot            uint32
	epoch, generation, value uint64
	body                     []byte
}

func readFrame(r io.Reader) (f frame, err error) {
	var h [4]byte
	if _, err = io.ReadFull(r, h[:]); err != nil {
		return
	}
	n := binary.LittleEndian.Uint32(h[:])
	if n < 33 || n > 65569 {
		return f, fmt.Errorf("private frame size %d", n)
	}
	b := make([]byte, n)
	if _, err = io.ReadFull(r, b); err != nil {
		return
	}
	f = frame{b[0], binary.LittleEndian.Uint32(b[1:]), binary.LittleEndian.Uint32(b[5:]), binary.LittleEndian.Uint64(b[9:]), binary.LittleEndian.Uint64(b[17:]), binary.LittleEndian.Uint64(b[25:]), b[33:]}
	return
}
func writeFrame(w io.Writer, f frame) error {
	b := make([]byte, 37+len(f.body))
	binary.LittleEndian.PutUint32(b, uint32(len(b)-4))
	b[4] = f.kind
	binary.LittleEndian.PutUint32(b[5:], f.request)
	binary.LittleEndian.PutUint32(b[9:], f.slot)
	binary.LittleEndian.PutUint64(b[13:], f.epoch)
	binary.LittleEndian.PutUint64(b[21:], f.generation)
	binary.LittleEndian.PutUint64(b[29:], f.value)
	copy(b[37:], f.body)
	for len(b) > 0 {
		n, e := w.Write(b)
		if e != nil {
			return e
		}
		if n == 0 {
			return io.ErrNoProgress
		}
		b = b[n:]
	}
	return nil
}

type key struct {
	slot       uint32
	generation uint64
}
type session struct {
	lease *nativewire.Lease

	root           *instance
	id             key
	conn           net.Conn
	mu             sync.Mutex
	cv             *sync.Cond
	sent, consumed uint64
	dead           bool
	output         chan frame
	ended          chan struct{}
}

func (s *session) close() {
	s.mu.Lock()
	if !s.dead {
		s.dead = true
		close(s.ended)
		s.conn.Close()
		s.cv.Broadcast()
	}
	s.mu.Unlock()
}

type instance struct {
	owner             *nativewire.Owner
	beforeRelayDone   func(*session, uint8)
	completionTarget  uint32
	completionEntered chan struct{}
	completionRelease <-chan struct{}
	completionOnce    sync.Once
	pipeReject        chan struct{}

	duplicateStop            bool
	receiptCount             int
	wrongFence, wrongStopAck bool
	disposed                 bool
	hostDone                 chan struct{}
	hostWaitOnce             sync.Once
	capture                  prepared.Capture
	approval                 chan error
	approvalOnce             sync.Once
	readDone                 chan struct{}
	checking, consumed       bool
	captureID                uint64
	stopRequest              uint32
	stopAcked, stopReceipt   bool
	lifecycleEvents          []string
	operation                chan struct{}
	dropKinds                map[byte]bool
	wrongReceipt             bool
	lease                    *forklimit.Lease
	captureDebt              error
	retained                 []*stored.Owned
	failAcquisition          bool
	afterFence               func(uint64) error

	epoch                      uint64
	in, guestIn, out, guestOut *os.File
	writeMu, mu                sync.Mutex
	pending                    map[uint32]chan frame
	opening                    map[uint32]*session
	sessions                   map[key]*session
	sequence                   atomic.Uint32
	ready                      chan struct{}
	poisoned                   chan struct{}
	done                       chan struct{}
	wg                         sync.WaitGroup
	stopping                   bool
	err, errorCause            error
	listener                   net.Listener
	limit                      uint64
	finalStats                 []byte
	events                     []frame
}

var epochCounter atomic.Uint64

func newInstance(ms uint64) (*instance, error) { return newRestored(ms, nil) }
func newRestored(ms uint64, restore []prepared.Entry) (*instance, error) {
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	return newRestoredContext(ctx, ms, restore)
}
func newRestoredContext(ctx context.Context, ms uint64, restore []prepared.Entry) (*instance, error) {
	epoch, e := nativewire.Next64(&epochCounter)
	if e != nil {
		return nil, e
	}
	own := nativewire.New(epoch)
	r := &instance{owner: own, pipeReject: make(chan struct{}, 1), epoch: epoch, pending: map[uint32]chan frame{}, opening: map[uint32]*session{}, sessions: map[key]*session{}, ready: make(chan struct{}), poisoned: make(chan struct{}), done: make(chan struct{}), limit: ms, approval: make(chan error, 1), hostDone: make(chan struct{}), readDone: make(chan struct{}), operation: make(chan struct{}, 1), dropKinds: map[byte]bool{}}

	r.guestIn, r.in, e = os.Pipe()
	if e != nil {
		return nil, e
	}
	r.out, r.guestOut, e = os.Pipe()
	if e != nil {
		return nil, errors.Join(e, r.guestIn.Close(), r.in.Close())
	}
	complete := generatedgo.StartWireInstance(r.guestIn, r.guestOut, os.Stderr, "", restore, false, func(cause error) { go func() { r.fail(cause); r.guestIn.Close() }() }, r.epoch, &generatedgo.WireLifecycle{Capture: &r.capture, Approval: r.approval, Observe: r.note, BeforeWorkerComplete: r.workerCompletion})
	r.wg.Add(1)
	go func() { defer r.wg.Done(); defer close(r.readDone); r.readLoop() }()
	go func() {
		e := <-complete
		r.mu.Lock()
		r.err = e
		r.mu.Unlock()
		if e != nil {
			r.fail(e)
		}
		r.guestIn.Close()
		r.guestOut.Close()
		close(r.done)
	}()
	select {
	case <-r.ready:
		return r, nil
	case <-r.done:
		return r, fmt.Errorf("runtime before ready: %v", r.result())
	case <-ctx.Done():
		r.fail(ctx.Err())
		r.guestIn.Close()
		return r, ctx.Err()
	}
}
func (r *instance) result() error {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.errorCause != nil {
		return r.errorCause
	}
	return r.err
}
func (r *instance) note(e string) {
	r.mu.Lock()
	r.lifecycleEvents = append(r.lifecycleEvents, e)
	r.mu.Unlock()
}
func (r *instance) resolve(e error) { r.approvalOnce.Do(func() { r.approval <- e }) }
func (r *instance) fail(e error) {
	r.resolve(e)
	r.mu.Lock()
	if r.errorCause == nil {
		r.errorCause = e
	}
	r.owner.StopLocked()
	all := make([]*session, 0, len(r.sessions))
	for _, s := range r.sessions {
		all = append(all, s)
	}
	r.mu.Unlock()
	for _, s := range all {
		s.close()
	}
}
func (r *instance) send(f frame) error {
	f.epoch = r.epoch
	r.writeMu.Lock()
	defer r.writeMu.Unlock()
	if f.kind == 12 && r.duplicateStop {
		var b bytes.Buffer
		writeFrame(&b, f)
		writeFrame(&b, f)
		_, e := io.Copy(r.in, &b)
		if e == nil {
			r.note("duplicate_stop_sent")
		}
		return e
	}
	return writeFrame(r.in, f)
}

type unsentControl struct{ error }

func (e *unsentControl) Unwrap() error { return e.error }

// Closing the peer read end aborts an uncertain/partial control write.
func (r *instance) sendContext(ctx context.Context, f frame) error {
	if e := ctx.Err(); e != nil {
		return &unsentControl{e}
	}
	done := make(chan error, 1)
	go func() { done <- r.send(f) }()
	select {
	case e := <-done:
		return e
	case <-ctx.Done():
		r.guestIn.Close()
		<-done
		return ctx.Err()
	}
}
func (r *instance) reserveLocked(c net.Conn) (*session, error) {
	l, e := r.owner.ReserveLocked()
	if e != nil {
		return nil, e
	}
	t := l.Token()
	s := &session{root: r, lease: l, id: key{t.Slot, t.Generation}, conn: c, output: make(chan frame, 16), ended: make(chan struct{})}
	s.cv = sync.NewCond(&s.mu)
	r.sessions[s.id] = s
	return s, nil
}
func (s *session) relayDone(part uint8) {
	r := s.root
	r.mu.Lock()
	hook := r.beforeRelayDone
	r.mu.Unlock()
	if hook != nil {
		hook(s, part)
	}
	r.mu.Lock()
	released, e := r.owner.RelayDoneLocked(s.lease, part)
	if released {
		delete(r.sessions, s.id)
	}
	r.mu.Unlock()
	if e != nil {
		r.fail(e)
	}
}
func (r *instance) nativeRetired(s *session) {
	s.close()
	r.mu.Lock()
	released, e := r.owner.NativeRetiredLocked(s.lease)
	if released {
		delete(r.sessions, s.id)
	}
	r.mu.Unlock()
	if e != nil {
		r.fail(e)
	}
}
func (r *instance) workerCompletion(td uint32) {
	r.mu.Lock()
	target, entered, release := r.completionTarget, r.completionEntered, r.completionRelease
	r.mu.Unlock()
	if td == target && release != nil {
		r.completionOnce.Do(func() { close(entered) })
		<-release
	}
}
func (r *instance) pipe() (net.Conn, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.stopping || r.checking || r.consumed || r.errorCause != nil {
		return nil, errors.New("source not accepting")
	}
	if r.owner.CountLocked() >= 16 {
		select {
		case r.pipeReject <- struct{}{}:
		default:
			return nil, nativewire.ErrCapacity
		}
		client, server := net.Pipe()
		r.wg.Add(1)
		go func() { defer r.wg.Done(); defer func() { <-r.pipeReject }(); mysqlwire.RejectCapacity(server) }()
		return client, nil
	}
	client, server := net.Pipe()
	s, e := r.reserveLocked(server)
	if e != nil {
		client.Close()
		server.Close()
		if errors.Is(e, nativewire.ErrExhausted) {
			r.errorCause = e
			go func() { r.fail(e); r.guestIn.Close() }()
		}
		return nil, e
	}
	r.wg.Add(1)
	go func() { defer r.wg.Done(); r.attach(s) }()
	return client, nil
}
func (r *instance) rpc(f frame, initial ...*session) (frame, error) {
	ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
	defer cancel()
	return r.rpcContext(ctx, f, initial...)
}
func (r *instance) rpcContext(ctx context.Context, f frame, initial ...*session) (frame, error) {
	id, identityErr := nativewire.Next32(&r.sequence)
	if identityErr != nil {
		r.fail(identityErr)
		r.guestIn.Close()
		return frame{}, identityErr
	}
	f.request = id
	ch := make(chan frame, 1)
	r.mu.Lock()
	r.pending[id] = ch
	if f.kind == 12 {
		r.stopRequest = id
	}
	if len(initial) > 0 {
		r.opening[id] = initial[0]
	}
	r.mu.Unlock()
	defer func() { r.mu.Lock(); delete(r.pending, id); delete(r.opening, id); r.mu.Unlock() }()
	if e := r.sendContext(ctx, f); e != nil {
		return frame{}, e
	}
	select {
	case answer := <-ch:
		return answer, nil
	case <-r.done:
		return frame{}, fmt.Errorf("runtime ended: %v", r.result())
	case <-ctx.Done():
		return frame{}, ctx.Err()
	}
}
func (r *instance) readLoop() {
	for {
		f, e := readFrame(r.out)
		if e != nil {
			r.resolve(fmt.Errorf("capture transport ended: %w", e))
			select {
			case <-r.done:
			default:
				r.mu.Lock()
				normal := r.stopping
				r.mu.Unlock()
				if !normal {
					r.fail(e)
				}
			}
			return
		}
		if f.epoch != r.epoch {
			r.fail(errors.New("wrong instance identity"))
			return
		}
		r.mu.Lock()
		drop := r.dropKinds[f.kind]
		wrong := r.wrongReceipt
		wrongFence, wrongAck := r.wrongFence, r.wrongStopAck
		r.mu.Unlock()
		if drop {
			continue
		}
		if (wrong && f.kind == 114) || (wrongFence && f.kind == 112) || (wrongAck && f.kind == 113) {
			f.value++
		}
		if f.kind == 114 {
			r.mu.Lock()
			valid := r.consumed && r.captureID != 0 && f.value == r.captureID && f.request == r.stopRequest
			acked := r.stopAcked
			if valid {
				r.stopReceipt = true
				r.receiptCount++
				r.lifecycleEvents = append(r.lifecycleEvents, "capture_stopped")
			}
			r.mu.Unlock()
			if !valid {
				r.fail(errors.New("wrong capture receipt"))
				continue
			}
			if acked {
				r.resolve(nil)
			}
			continue
		}
		r.mu.Lock()
		if len(r.events) < 10000 {
			r.events = append(r.events, frame{kind: f.kind, slot: f.slot, generation: f.generation, value: f.value})
		}
		if f.kind == opened && r.opening[f.request] != nil {
			initial := r.opening[f.request]
			if e := r.owner.OpenedLocked(initial.lease, nativewire.Token{r.epoch, f.slot, f.generation}); e != nil {
				r.mu.Unlock()
				r.fail(e)
				r.guestIn.Close()
				return
			}
		}
		initial := r.opening[f.request]
		if f.kind == 110 && initial != nil {
			if f.value != 4 || r.owner.RejectLocked(initial.lease, nativewire.Token{r.epoch, f.slot, f.generation}) != nil {
				r.mu.Unlock()
				r.fail(errors.New("unexpected native OPEN rejection"))
				r.guestIn.Close()
				return
			}
		}
		ch := r.pending[f.request]
		s := r.sessions[key{f.slot, f.generation}]
		stale := r.owner.StaleLocked(nativewire.Token{r.epoch, f.slot, f.generation})
		r.mu.Unlock()
		if ch != nil {
			ch <- f
			continue
		}
		switch f.kind {
		case ready:
			var profile struct {
				Lifecycle int `json:"lifecycle_profile"`
			}
			if json.Unmarshal(f.body, &profile) != nil || profile.Lifecycle != 2 {
				r.fail(errors.New("wrong lifecycle profile"))
				r.guestIn.Close()
				return
			}
			close(r.ready)
		case raw:
			if s == nil {
				if stale {
					continue
				}
				r.fail(errors.New("output without session"))
				return
			}
			select {
			case s.output <- f:
			case <-s.ended:
			default:
				r.fail(errors.New("bounded host output queue overflow"))
				return
			}
		case credit:
			if s != nil {
				s.mu.Lock()
				s.consumed = f.value
				s.cv.Broadcast()
				s.mu.Unlock()
			}
		case retired:
			if s != nil {
				r.nativeRetired(s)
			}
		case 111:
			r.mu.Lock()
			r.finalStats = append([]byte(nil), f.body...)
			receipt := r.stopReceipt
			r.mu.Unlock()
			if r.capture.Requested() && !receipt {
				r.fail(errors.New("missing capture receipt"))
			}
		case poison:
			r.fail(fmt.Errorf("guest poison: %s", f.body))
			close(r.poisoned)
		}
	}
}
func (r *instance) serve() (string, error) {
	l, e := net.Listen("tcp", "127.0.0.1:0")
	if e != nil {
		return "", e
	}
	r.listener = l
	r.wg.Add(1)
	go func() {
		defer r.wg.Done()
		for {
			c, e := l.Accept()
			if e != nil {
				return
			}
			if tcp, ok := c.(*net.TCPConn); ok {
				tcp.SetWriteBuffer(1024)
			}
			r.mu.Lock()
			deny := r.stopping || r.checking || r.consumed || r.errorCause != nil
			var session *session
			var reservationErr error
			if !deny {
				session, reservationErr = r.reserveLocked(c)
			}
			full := errors.Is(reservationErr, nativewire.ErrCapacity)
			if !deny && reservationErr == nil {
				r.wg.Add(1)
			}
			r.mu.Unlock()
			if deny {
				c.Close()
				continue
			}
			if full {
				// Inline in the one accept owner, outside mu; fixed1s helper.
				mysqlwire.RejectCapacity(c)
				continue
			}
			if reservationErr != nil {
				c.Close()
				if errors.Is(reservationErr, nativewire.ErrExhausted) {
					r.fail(reservationErr)
					r.guestIn.Close()
				}
				continue
			}
			go func() { defer r.wg.Done(); r.attach(session) }()
		}
	}()
	return l.Addr().String(), nil
}
func (r *instance) attach(s *session) {
	c := s.conn
	writerStarted := false
	defer s.relayDone(nativewire.Reader)
	defer func() {
		if !writerStarted {
			s.relayDone(nativewire.Writer)
		}
	}()
	r.mu.Lock()
	e := r.owner.BeginOpenLocked(s.lease)
	if e != nil {
		r.owner.CancelReservedLocked(s.lease)
	}
	r.mu.Unlock()
	if e != nil {
		s.close()
		return
	}
	f, e := r.rpc(frame{kind: open, slot: s.id.slot, generation: s.id.generation, value: r.limit}, s)
	if e != nil {
		// A lost OPEN reply is uncertain: retain the reservation until disposal.
		r.mu.Lock()
		r.owner.UncertainLocked(s.lease)
		r.mu.Unlock()
		r.fail(e)
		c.Close()
		return
	}
	if f.kind != opened {
		s.close()
		if f.kind == 110 && f.value == 3 {
			mysqlwire.RejectCapacity(c)
		} else {
			c.Close()
		}
		return
	}
	r.mu.Lock()
	closing := r.stopping
	if !closing {
		r.wg.Add(1)
	}
	r.mu.Unlock()
	if closing {
		s.close()
		return
	}
	writerStarted = true
	go func() {
		defer r.wg.Done()
		defer s.relayDone(nativewire.Writer)
		defer s.close()
		for {
			select {
			case f := <-s.output:
				b := f.body
				for len(b) > 0 {
					n, e := c.Write(b)
					if e != nil {
						r.send(frame{kind: peerEOF, slot: s.id.slot, generation: s.id.generation, value: 1})
						return
					}
					b = b[n:]
				}
				if e := r.send(frame{kind: ack, slot: s.id.slot, generation: s.id.generation, value: f.value + uint64(len(f.body))}); e != nil {
					return
				}
			case <-s.ended:
				return
			}
		}
	}()
	defer func() {
		r.mu.Lock()
		stopping := r.stopping
		r.mu.Unlock()
		if !stopping {
			r.send(frame{kind: peerEOF, slot: s.id.slot, generation: s.id.generation})
		}
	}()
	b := make([]byte, 4096)
	for {
		s.mu.Lock()
		for !s.dead && s.sent-s.consumed > inputLimit {
			s.cv.Wait()
		}
		if s.dead {
			s.mu.Unlock()
			return
		}
		// One bounded observation byte beyond the credit window lets native
		// input() decide overflow under the same gate as capture. It is never
		// appended above the guest queue limit. EOF stays observable at full.
		available := uint64(inputLimit) - (s.sent - s.consumed) + 1
		offset := s.sent
		s.mu.Unlock()
		size := len(b)
		if uint64(size) > available {
			size = int(available)
		}
		n, e := c.Read(b[:size])
		if n > 0 {
			s.mu.Lock()
			s.sent += uint64(n)
			s.mu.Unlock()
			if sendErr := r.send(frame{kind: data, slot: s.id.slot, generation: s.id.generation, value: offset, body: b[:n]}); sendErr != nil {
				return
			}
		}
		if e != nil {
			return
		}
	}
}
func (r *instance) close() error {
	r.resolve(errors.New("ordinary close cannot approve capture"))
	r.mu.Lock()
	r.stopping = true
	r.owner.StopLocked()
	l := r.listener
	all := make([]*session, 0, len(r.sessions))
	for _, s := range r.sessions {
		all = append(all, s)
	}
	r.mu.Unlock()
	if l != nil {
		l.Close()
	}
	for _, s := range all {
		s.close()
	}
	go r.send(frame{kind: stop})
	select {
	case <-r.done:
		r.in.Close()
		r.wg.Wait()
		r.mu.Lock()
		if r.owner != nil {
			r.owner.DisposeLocked()
		}
		r.sessions = map[key]*session{}
		r.mu.Unlock()
		for _, o := range r.retained {
			o.Close()
		}
		r.retained = nil
		if r.lease != nil {
			r.lease.Settle(r.captureDebt)
		}
		r.out.Close()
		return errors.Join(r.result(), r.captureDebt)
	case <-time.After(12 * time.Second):
		e := errors.New("join incomplete; running resources retained")
		if r.lease != nil {
			r.lease.Settle(e)
		}
		return e
	}
}
func (r *instance) stopIngress() {
	r.mu.Lock()
	r.stopping = true
	r.owner.StopLocked()
	l := r.listener
	all := make([]*session, 0, len(r.sessions))
	for _, s := range r.sessions {
		all = append(all, s)
	}
	r.mu.Unlock()
	if l != nil {
		l.Close()
	}
	for _, s := range all {
		s.close()
	}
}
func (r *instance) awaitJoined(ctx context.Context) error {
	select {
	case <-r.done:
	case <-ctx.Done():
		if r.lease != nil {
			r.lease.Settle(ctx.Err())
		}
		return fmt.Errorf("join incomplete; resources retained: %w", ctx.Err())
	}
	select {
	case <-r.readDone:
	case <-ctx.Done():
		return fmt.Errorf("reader join incomplete: %w", ctx.Err())
	}
	r.hostWaitOnce.Do(func() { go func() { r.wg.Wait(); close(r.hostDone) }() })
	select {
	case <-r.hostDone:
	case <-ctx.Done():
		if r.lease != nil {
			r.lease.Settle(ctx.Err())
		}
		return fmt.Errorf("host join incomplete: %w", ctx.Err())
	}
	r.in.Close()
	r.out.Close()
	r.mu.Lock()
	r.disposed = true
	if r.owner != nil {
		r.owner.DisposeLocked()
	}
	r.sessions = map[key]*session{}
	r.mu.Unlock()
	r.note("host_joined")
	return r.result()
}
func (r *instance) failCapture(ctx context.Context, e error) (*stored.Owned, bool, error) {
	r.mu.Lock()
	r.consumed = true
	r.checking = false
	r.owner.EndCaptureLocked()
	r.mu.Unlock()
	r.capture.Discard()
	r.fail(e)
	r.stopIngress()
	go r.send(frame{kind: stop})
	joined := r.awaitJoined(ctx)
	r.mu.Lock()
	disposed := r.disposed
	r.mu.Unlock()
	if disposed && r.lease != nil {
		r.lease.Settle(r.captureDebt)
	}
	return nil, true, errors.Join(e, joined)
}
func (r *instance) snapshot(ctx context.Context, rollback bool) (*stored.Owned, bool, error) {
	select {
	case r.operation <- struct{}{}:
		defer func() { <-r.operation }()
	case <-ctx.Done():
		return nil, false, ctx.Err()
	}
	r.mu.Lock()
	if r.stopping || r.consumed || r.errorCause != nil {
		r.mu.Unlock()
		return nil, true, errors.New("source consumed or failed")
	}
	if e := r.owner.BeginCaptureLocked(); e != nil {
		r.mu.Unlock()
		return nil, false, e
	}
	r.checking = true
	captureSequence, identityErr := nativewire.Next32(&r.sequence)
	if identityErr != nil {
		r.mu.Unlock()
		r.fail(identityErr)
		r.guestIn.Close()
		return nil, true, identityErr
	}
	r.captureID = uint64(captureSequence)
	id := r.captureID
	r.mu.Unlock()
	if e := ctx.Err(); e != nil {
		r.mu.Lock()
		r.checking = false
		r.owner.EndCaptureLocked()
		r.mu.Unlock()
		return nil, false, e
	}
	flag := uint64(0)
	if rollback {
		flag = 1
	}
	reply, e := r.rpcContext(ctx, frame{kind: 11, value: id, generation: flag})
	if e != nil {
		var unsent *unsentControl
		if errors.As(e, &unsent) {
			r.mu.Lock()
			r.checking = false
			r.owner.EndCaptureLocked()
			r.mu.Unlock()
			return nil, false, e
		}
		return r.failCapture(ctx, e)
	}
	if reply.kind == 110 && (reply.value == 1 || reply.value == 2) {
		r.mu.Lock()
		r.checking = false
		r.owner.EndCaptureLocked()
		r.mu.Unlock()
		return nil, false, fmt.Errorf("capture precondition %d: %s", reply.value, reply.body)
	}
	if reply.kind != 112 || reply.value != id {
		return r.failCapture(ctx, errors.New("invalid capture fence"))
	}
	r.mu.Lock()
	r.consumed = true
	r.mu.Unlock()
	r.note("fenced")
	if r.afterFence != nil {
		if e = r.afterFence(id); e != nil {
			return r.failCapture(ctx, e)
		}
	}
	r.capture.Request()
	reply, e = r.rpcContext(ctx, frame{kind: 12, value: id})
	if e != nil {
		return r.failCapture(ctx, e)
	}
	if reply.kind != 113 || reply.value != id {
		return r.failCapture(ctx, errors.New("invalid capture stop acceptance"))
	}
	if e = ctx.Err(); e != nil {
		return r.failCapture(ctx, e)
	}
	r.mu.Lock()
	r.stopAcked = true
	r.lifecycleEvents = append(r.lifecycleEvents, "stop_accepted")
	receipt := r.stopReceipt
	r.mu.Unlock()
	if receipt {
		r.resolve(nil)
	}
	r.stopIngress()
	if e = r.awaitJoined(ctx); e != nil {
		return r.failCapture(ctx, e)
	}
	image, e := r.capture.TakeRoot(generatedgo.GuestSHA256)
	if e != nil {
		return nil, true, e
	}
	owned, e := stored.OwnRoot(image, generatedgo.GuestSHA256)
	if e != nil {
		return nil, true, e
	}
	r.note("root_owned")
	if r.failAcquisition {
		r.captureDebt = errors.New("injected captured-root cleanup failure")
		r.retained = append(r.retained, owned)
		if r.lease != nil {
			r.lease.Settle(r.captureDebt)
		}
		return nil, true, r.captureDebt
	}
	if r.lease != nil {
		r.lease.Settle(nil)
	}
	return owned, true, nil
}

type baseline struct {
	backing *stored.Owned
	pool    *forklimit.Pool
}

func makeBaseline(o *stored.Owned) *baseline {
	p, e := forklimit.New(2)
	if e != nil {
		panic(e)
	}
	return &baseline{o, p}
}
func (b *baseline) fork(ctx context.Context) (*instance, error) {
	lease, e := b.pool.Acquire(ctx)
	if e != nil {
		return nil, e
	}
	entries, release, e := b.backing.Acquire(generatedgo.GuestSHA256)
	if e != nil {
		lease.Cancel()
		return nil, e
	}
	defer release()
	if e = lease.Activate(); e != nil {
		return nil, e
	}
	r, e := newRestoredContext(ctx, 5000, entries)
	if r != nil {
		r.lease = lease
	}
	if e != nil {
		if r != nil {
			cleanupCtx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			joined := r.awaitJoined(cleanupCtx)
			cancel()
			r.mu.Lock()
			disposed := r.disposed
			r.mu.Unlock()
			if joined != nil && !disposed {
				lease.Settle(joined)
			} else {
				lease.Settle(nil)
			}
		} else {
			lease.Settle(e)
		}
		return r, e
	}
	return r, nil
}
func (b *baseline) close() error { b.pool.Close(); return b.backing.Close() }
