// Isolated side control for inspecting a live generated guest. Not a public API.
package main

import (
	"context"
	"encoding/json"
	"example.com/mariamem-spike/generated/base"
	"net"
	"net/http"
	"os"
	"path/filepath"
	"sync"
	"time"
)

func cowControl(m *base.Module, fs *base.MemFS) func() {
	dir := os.Getenv("COW_CONTROL_DIR")
	if dir == "" {
		return func() {}
	}
	ln, err := net.Listen("tcp4", "127.0.0.1:0")
	if err != nil {
		panic(err)
	}
	start := make(chan struct{})
	var once sync.Once
	mux := http.NewServeMux()
	mux.HandleFunc("/start", func(w http.ResponseWriter, r *http.Request) { once.Do(func() { close(start) }); w.Write([]byte("OK")) })
	mux.HandleFunc("/sample", func(w http.ResponseWriter, r *http.Request) { json.NewEncoder(w).Encode(base.SpikeInspect(m, fs)) })
	server := &http.Server{Handler: mux}
	go server.Serve(ln)
	b, _ := json.Marshal(map[string]any{"port": ln.Addr().(*net.TCPAddr).Port, "pid": os.Getpid()})
	if err = os.WriteFile(filepath.Join(dir, "control.json"), b, 0600); err != nil {
		panic(err)
	}
	if os.Getenv("COW_PAUSE_INIT") == "1" {
		<-start
	}
	return func() {
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		if err := server.Shutdown(ctx); err != nil {
			panic(err)
		}
	}
}
