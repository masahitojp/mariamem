const B = 16384
const identity = "d-component-fixture-v1"
type Ref struct { Seg int `json:"seg"`; Off int `json:"off"` }

type Store struct { mode string; f *os.File; mapped []byte; heap [][]byte; used int; capacity int; path string; packBytes int; frozen bool }
func newStore(mode,path string,capacity int)(*Store,error){
 s:=&Store{mode:mode,path:path,capacity:capacity};if mode!="d1"&&mode!="d2"{return nil,errors.New("mode")}

}
func(s *Store) alloc()(int,error){

}
func(s *Store) block(i int)[]byte{if s.mode=="d1"{return s.heap[i]};return s.mapped[i*B:(i+1)*B]}
func(s *Store) close()error{var e error;if s.mapped!=nil{e=syscall.Munmap(s.mapped);s.mapped=nil};if s.f!=nil{e=errors.Join(e,s.f.Close());s.f=nil};return e}
func(s *Store) freeze()error{
 if s.frozen{return errors.New("already frozen")};s.frozen=true
 if s.mode=="d1" {var e error;s.f,e=os.OpenFile(s.path,os.O_CREATE|os.O_EXCL|os.O_RDWR,0600);if e!=nil{return e};for _,b:=range s.heap{n,e:=s.f.Write(b);s.packBytes+=n;if e!=nil{return e};if n!=len(b){return io.ErrShortWrite};if os.Getenv("D_FAULT")=="transfer"{return errors.New("injected transfer")}}} else {if e:=syscall.Munmap(s.mapped);e!=nil{return e};s.mapped=nil;if e:=s.f.Truncate(int64(s.used));e!=nil{return e};if os.Getenv("D_FAULT")=="transfer"{return errors.New("injected transfer")}}

}
type FS struct { files map[string]*File; store *Store; parents [][]byte; ownSeg int; copied int }
func newFS(s *Store,parents [][]byte,m Manifest)*FS{
 fs:=&FS{files:map[string]*File{},store:s,parents:parents,ownSeg:len(parents)}
 for name,f:=range m.Files{v:=f;v.Refs=append([]Ref(nil),f.Refs...);fs.files[name]=&v};return fs
}
func(fs *FS) refBytes(r Ref)[]byte{if r.Seg<0{return nil};if r.Seg==fs.ownSeg{return fs.store.block(r.Off/B)};return fs.parents[r.Seg][r.Off:r.Off+B]}
func(fs *FS) resize(name string,size int)error{
 if fs.store.frozen{return errors.New("frozen")}

 if n<len(f.Refs){f.Refs=f.Refs[:n]};for len(f.Refs)<n{f.Refs=append(f.Refs,Ref{Seg:-1})};f.Size=size;return nil
}
func(fs *FS) own(f *File,i int,full bool)error{
 r:=f.Refs[i];if r.Seg==fs.ownSeg{return nil};slot,e:=fs.store.alloc();if e!=nil{return e}

}
func(fs *FS) write(name string,p []byte,off int)error{
 if fs.store.frozen{return errors.New("frozen")};if off<0||off>1<<30||len(p)>(1<<30)-off{return errors.New("write range")};if len(p)==0{return nil}

}
func(fs *FS) read(name string,p []byte,off int)(int,error){
 f:=fs.files[name];if f==nil{return 0,os.ErrNotExist};if off<0{return 0,errors.New("offset")};if len(p)==0{return 0,nil};if off>=f.Size{return 0,io.EOF};count:=min(len(p),f.Size-off);done:=0

}
func(fs *FS) manifest()Manifest{

func pattern(block int)[]byte {b:=make([]byte,B);for j:=range b{b[j]=byte((block*17+j*13+7)%251)};return b}
func seed(fs *FS,blocks,stride int)error{

func stats(s *Store,fs *FS)map[string]any{var m runtime.MemStats;runtime.ReadMemStats(&m);return map[string]any{"materialized_bytes":s.used,"reserved_capacity_bytes":s.capacity,"heap_payload_bytes":len(s.heap)*B,"pack_bytes":s.packBytes,"parent_copy_bytes":fs.copied,"total_alloc_bytes":m.TotalAlloc,"num_gc":m.NumGC}}
