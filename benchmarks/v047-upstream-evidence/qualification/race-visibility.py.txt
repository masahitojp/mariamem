import json, os, subprocess
from pathlib import Path
root=Path.cwd(); work=root.parent; d=work/'temp/race-visibility'; d.mkdir(exist_ok=True)
(d/'go.mod').write_text('module example.com/racevisibility\n\ngo 1.26.0\n')
source='''package visibility
import("os";"sync";"syscall";"testing")
func TestVisibility(t *testing.T) {
 b:=make([]byte,65536)
 if os.Getenv("BACKING")=="mmap" {var e error;b,e=syscall.Mmap(-1,0,65536,syscall.PROT_READ|syscall.PROT_WRITE,syscall.MAP_PRIVATE|syscall.MAP_ANON);if e!=nil {t.Fatal(e)};defer syscall.Munmap(b)}
 var w sync.WaitGroup;w.Add(2)
 for k:=0;k<2;k++ {go func(){defer w.Done();for i:=0;i<10000;i++ {b[0]++}}()};w.Wait()
}
'''
(d/'visibility_test.go').write_text(source)
(work/'evidence/race-visibility.go.txt').write_text(source)
result={}
for backing in ['heap','mmap']:
 p=subprocess.run(['go','test','-race','-count=1','.'],cwd=d,env=dict(os.environ,BACKING=backing),text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
 (work/'evidence'/f'race-visibility-{backing}.log').write_text(p.stdout)
 result[backing]={'returncode':p.returncode,'race_reported':'WARNING: DATA RACE' in p.stdout}
(work/'evidence/race-visibility.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
assert result['heap']['race_reported'] and not result['mmap']['race_reported']
